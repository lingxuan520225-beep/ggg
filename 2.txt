/* ---------- 存档 ---------- */
function maiData(){
  if(!S.mai) S.mai = {};
  const m = S.mai;
  if(!m.small) m.small = {};
  if(!m.small.fr)  m.small.fr = {};
  if(!m.small.lmg) m.small.lmg = {};
  if(!m.big)  m.big = {};
  if(!m.mine) m.mine = {};
  if(!m.reroll) m.reroll = {};
  m.ore = Math.max(0, Math.floor(m.ore || 0));
  m.ly  = Math.max(0, Math.floor(m.ly  || 0));
  m.mineTs = m.mineTs || 0;
  return m;
}
function maiResName(realm){ return realm === 'lmg' ? '灵蕴' : '矿石'; }

/* ---------- 小节点 ---------- */
function maiSmallDef(realm, k){
  const list = MAI_SMALL[realm] || [];
  for(let i = 0; i < list.length; i++) if(list[i].k === k) return list[i];
  return null;
}
function maiSmallName(realm, k){ const d = maiSmallDef(realm, k); return d ? d.n : k; }
/* 1.94：小节点取消等级上限（原 300）。改为「高投入高回报」——每级加成 ×2、成本指数 1.25→1.5、
   基价 ×1.5，靠成本曲线自然收敛，不再硬性夹断。 */
const MAI_SMALL_MAX_LV = null;
function _maiClampS(v){
  const x = Math.max(0, Math.floor(v || 0));
  return MAI_SMALL_MAX_LV ? Math.min(MAI_SMALL_MAX_LV, x) : x;
}
function maiSmallLv(realm, k){
  const m = maiData();
  return _maiClampS((m.small[realm] || {})[k] || 0);
}
function maiSmallSet(realm, k, v){ maiData().small[realm][k] = _maiClampS(v); }
/* 成本：幂函数递增（延续 1.41 的偏好，前高后缓）；1.49 下调（小节点便宜） */
/* 按「当前等级」算价的纯函数 —— 批量预估需要在不真实扣资源的前提下试算多级，
   故与 maiSmallCost 拆开：后者读实时等级，前者可传入任意 lv。 */
function maiSmallCostAt(realm, lv){
  const base = realm === 'lmg' ? 36 : 108;     /* 1.96：基价 30/90 → 36/108（成本 +20%） */
  return Math.round(base * Math.pow(lv + 1, 1.43));  /* 1.96：指数 1.4 → 1.43 */
}
function maiSmallCost(realm, k){ return maiSmallCostAt(realm, maiSmallLv(realm, k)); }
function maiSmallSum(realm, k){
  const d = maiSmallDef(realm, k);
  return d ? d.per * maiSmallLv(realm, k) : 0;
}

/* ---------- 大节点 ---------- */
function maiBigDef(k){ for(let i = 0; i < MAI_BIG.length; i++) if(MAI_BIG[i].k === k) return MAI_BIG[i]; return null; }
function maiBigLv(k, realm){
  const m = maiData();
  const d = m.big[k] || (m.big[k] = {fr:0, lmg:0});
  const v = Math.max(0, Math.floor(d[realm] || 0));
  /* 1.94：读回时 clamp 到「绑定小节点 ÷ 10」——旧档曾按固定上限存过值，
     解除上限后若小节点等级偏低，残留等级会超出绑定上限，需在此收敛。 */
  if(!maiBigDef(k)) return v;
  const mx = maiBigMaxLv(k, realm);
  return mx > 0 ? Math.min(mx, v) : v;
}
function maiBigSet(k, realm, v){
  const m = maiData();
  if(!m.big[k]) m.big[k] = {fr:0, lmg:0};
  const x = Math.max(0, Math.floor(v));
  m.big[k][realm] = MAI_BIG_MAX ? Math.min(MAI_BIG_MAX, x) : x;
}
/* 该界该大节点的等级上限 = 绑定小节点等级 ÷ 10 */
function maiBigMaxLv(k, realm){
  const d = maiBigDef(k); if(!d) return 0;
  const sk = realm === 'lmg' ? d.sl : d.sf;
  const n = Math.floor(maiSmallLv(realm, sk) / MAI_SMALL_PER_BIG);
  return MAI_BIG_MAX ? Math.min(MAI_BIG_MAX, n) : n;
}
function maiBigCostAt(realm, lv){
  const base = realm === 'lmg' ? 276 : 840;      /* 1.96：基价 230/700 → 276/840（成本 +20%） */
  return Math.round(base * Math.pow(lv + 1, 1.88));   /* 1.96：指数 1.85 → 1.88 */
}
function maiBigCost(k, realm){ return maiBigCostAt(realm, maiBigLv(k, realm)); }
/* 单界加成 %：属性类 ×MAI_BIG_SOLO_ATTR，资源类 ×MAI_BIG_SOLO */
function maiBigSolo(k, realm){
  const d = maiBigDef(k); if(!d) return 0;
  const per = d.res ? MAI_BIG_SOLO : MAI_BIG_SOLO_ATTR;
  return per * maiBigLv(k, realm);
}
/* 共鸣：取两界较小值（必须两界都投入才涨） */
function maiBigReso(k){ return Math.min(maiBigLv(k, 'fr'), maiBigLv(k, 'lmg')); }
/* 共鸣等级 = 所有大节点（有上限者）的共鸣总和，0 ~ 8×1000 = 8000 */
function maiReso(){
  let s = 0;
  for(let i = 0; i < MAI_BIG.length; i++) s += maiBigReso(MAI_BIG[i].k);
  return s;
}
function maiResoMax(){ return MAI_BIG_MAX ? MAI_BIG.length * MAI_BIG_MAX : null; }
function maiResoPct(){ return maiReso() * MAI_RESO_PER; }
/* 1.75：共鸣对「灵界属性」改为独立乘区（不再进加法池被稀释）。
   仅作用于 lin / exp / fix 类战斗属性，pct 类（暴击/暴伤/连击/资源）不享受，
   与改之前的作用范围保持一致。 */
function maiResoMul(){ return 1 + maiResoPct() / 100; }

/* ---------- 加成出口（供其它系统调用） ---------- */

/* ---------- 矿脉：每日刷新 ---------- */
function maiMaxMineLv(realm){
  const lv = realm === 'lmg' ? (S.lmgMaxLv || 1) : (S.stLv || 1);
  return Math.max(1, Math.floor(lv / 10));
}
function maiRollQ(){
  const w = MAI_Q_W;
  let r = Math.random() * 100, acc = 0, q = 0;
  for(let i = 0; i < w.length; i++){ acc += w[i]; if(r < acc){ q = i; break; } q = i; }
  return q;
}
function maiRollLv(maxLv){
  if(maxLv <= 1) return 1;
  return Math.max(1, Math.min(maxLv, Math.ceil(Math.pow(Math.random(), 2.2) * maxLv)));
}
function maiMineList(realm){
  const m = maiData();
  if(!m.mine[realm] || !Array.isArray(m.mine[realm])) m.mine[realm] = [];
  return m.mine[realm];
}
function maiMineRoll(realm){
  const m = maiData();
  const maxLv = maiMaxMineLv(realm);
  const arr = [];
  for(let i = 0; i < MAI_MINE_N; i++){
    arr.push({q: maiRollQ(), lv: maiRollLv(maxLv), occ: false, st: 0, acc: 0});
  }
  /* 1.97 高品质保底（语义修正）：不再是「每批保证 N 处」，而是「连续 N 次刷新都没出高品质
     才补偿 1 处」—— 中途只要出了任意一处即重新计数，故高品质仍以权重为主、稀有度得以保留。
     补位只升 1 处且取品质最低者，默认给神品，仅 MAI_MINE_HI_XIAN 概率直接给仙品。 */
  let hi = 0;
  for(let i = 0; i < arr.length; i++) if(arr[i].q >= 3) hi++;
  if(!m.hiMiss) m.hiMiss = {};
  if(hi > 0){
    m.hiMiss[realm] = 0;                        /* 本批出了 → 连续未出计数归零 */
  } else {
    m.hiMiss[realm] = (m.hiMiss[realm] || 0) + 1;
    if(m.hiMiss[realm] >= MAI_MINE_HI_MISS){
      m.hiMiss[realm] = 0;
      let low = 0;
      for(let i = 1; i < arr.length; i++) if(arr[i].q < arr[low].q) low = i;
      arr[low].q = (Math.random() < MAI_MINE_HI_XIAN) ? 4 : 3;
    }
  }
  m.mine[realm] = arr;
}
/* 每满 24 小时重新随机全部矿点，并清除占领（需重新打） */
const MAI_REFRESH_MS = 24 * 3600 * 1000;
function maiDailyRoll(){
  const m = maiData();
  const now = Date.now();
  if(m.mineTs && (now - m.mineTs) < MAI_REFRESH_MS) return false;
  m.mineTs = now;
  maiMineRoll('fr'); maiMineRoll('lmg');
  m.reroll.fr = 0; m.reroll.lmg = 0;
  return true;
}
/* 占领该矿所需的最高关卡 */
function maiGuard(realm, q, lv){
  return Math.max(1, Math.ceil(lv * 10 * (MAI_Q_NEED[q] || 1)));
}
function maiCurLv(realm){ return Math.max(1, realm === 'lmg' ? (S.lmgMaxLv || 1) : (S.stLv || 1)); }
/* ---------- 矿脉守卫战：占领必须击败守卫 ----------
   凡矿 → 凡人界战斗数值；灵矿 → 灵界战斗数值（与用户要求一致）。
   守卫血量 = 该界第（矿等级 × 10）关怪物血量 × 品质倍率 —— 随玩家进度自动缩放，
   且「等级越高、品质越高」都让血量成倍上升，二者叠加即为「超级厉害」。
   守卫攻击 = 玩家该界最大血量的百分比 × (1 + 等级×5%) —— 同样随品质与等级上升，
   且因按玩家血量取比例，任何阶段都不会出现「一击秒杀」或「刮痧打不动」。 */
const MAI_GUARD_HP_MUL = [1, 2, 4, 8, 16];
const MAI_GUARD_ATK_Q  = 0.02;      /* 基础：玩家最大血 2%，每档品质 +1% */
const MAI_GUARD_ATK_LV = 0.05;      /* 每级 +5% */
const MAI_GUARD_ATK_MS = 2000;      /* 守卫每 2 秒反击一次 */

function maiGuardMaxHp(realm, q, lv){
  const L = Math.max(1, lv * 10);
  let base;
  if(realm === 'lmg'){
    base = dNum(lmgMonHpFor(L));
  } else {
    base = Number(dNum(dmul(BASE_HP, dpow(BASE_GROW, L - 1))));   /* BASE_HP / BASE_GROW 本身即大数对象 */
  }
  if(!isFinite(base) || base <= 0) base = 1e4;
  return base * MAI_GUARD_HP_MUL[q];
}
function maiGuardAtk(realm, q, lv){
  let mx;
  if(realm === 'lmg') mx = dNum(lmgAttrD('hp'));
  else mx = dNum(_toD(S.maxHp || D(1)));
  if(!isFinite(mx) || mx <= 0) mx = 1e3;
  return Math.max(1, mx * (MAI_GUARD_ATK_Q + 0.01 * q) * (1 + lv * MAI_GUARD_ATK_LV));
}
/* 放弃该界占领（可选：先结算已累积产出，不让玩家白等） */
function maiAbandon(realm, keepAcc){
  const list = maiMineList(realm);
  let n = 0;
  for(let i = 0; i < list.length; i++){
    const mn = list[i]; if(!mn || !mn.occ) continue;
    if(!keepAcc){
      const got = Math.floor(maiAcc(realm, mn));
      if(got > 0){
        const m = maiData();
        if(realm === 'lmg') m.ly  = Math.min(SAFE_INT, m.ly  + got);
        else                m.ore = Math.min(SAFE_INT, m.ore + got);
      }
    }
    mn.occ = false; mn.acc = 0; mn.st = 0; n++;
  }
  if(S.inMine && S.inMine.realm === realm) S.inMine = null;
  return n;
}
/* 挑战：进入守卫战 */
function maiMineOccupy(realm, i){
  const list = maiMineList(realm);
  const mn = list[i]; if(!mn || mn.occ) return;
  if(S.realm !== realm){
    toast('需先' + (realm === 'lmg' ? '飞升进入灵界' : '返回凡人界') + '方可挑战');
    return;
  }
  if(S.inTower || S.inBoss || S.inSecret){ toast('请先退出当前挑战'); return; }
  /* 每界同时只能占领 1 处：进入战斗前先放弃已占领的（产出先结算，不白等） */
  const ab = maiAbandon(realm, false);
  if(ab) toast('已放弃原有矿脉，产出已结算');
  const hp = maiGuardMaxHp(realm, mn.q, mn.lv);
  S.inMine = {realm: realm, i: i, hp: hp, maxHp: hp, last: Date.now()};
  applyMineUI(); renderMai();
  logMsg('灵脉·挑战' + (realm === 'lmg' ? '灵矿' : '凡矿') + '守卫（' + MAI_Q_N[mn.q] + '品 Lv.' + mn.lv + '）');
  toast('击败守卫即可占领！');
  save();
}

  if(Math.random() * 100 < calcCrit()) sum = dmul(sum, D(critMult()));
  sum = dadd(sum, dmul(_atk, D(calcIgn() / 100)));
  { const _cs = {n:0, hit:false}; sum = comboAdd(sum, _atk, 0.15, 0.92, 0.16, _cs); }
  sum = dmul(sum, D(1 + tSum('nmult') / 100));
  sum = dmul(sum, D(1 + calcFdm() / 100));
  return sum;
}

function doMineHit(manual){
  const f = S.inMine; if(!f) return null;
  if(manual && typeof manualClickOk === 'function' && !manualClickOk()) return null;
  hitInc();
  const list = maiMineList(f.realm), mn = list[f.i];
  if(!mn){ S.inMine = null; applyMineUI(); return null; }
  const dmg = (f.realm === 'lmg') ? mineLmgDmg() : dNum(mineFrDmg());
  f.hp = Math.max(0, f.hp - (isFinite(dmg) ? dmg : 0));
  /* 守卫反击 */
  const now = Date.now();
  if(now - (f.last || 0) >= MAI_GUARD_ATK_MS){
    f.last = now;
    const atk = maiGuardAtk(f.realm, mn.q, mn.lv);
    if(f.realm === 'lmg'){
      const h = dsub(_toD(S.lmgHp || D(0)), D(atk));
      S.lmgHp = dcmp(h, D(0)) < 0 ? D(0) : h;
    } else {
      const h = dsub(_toD(S.hp || D(0)), D(atk));
      S.hp = dcmp(h, D(0)) < 0 ? D(0) : h;
    }
  }
  /* 败：血量归零 */
  const dead = (f.realm === 'lmg') ? dcmp(_toD(S.lmgHp || D(0)), D(1)) < 0
                                   : dcmp(_toD(S.hp || D(0)), D(1)) < 0;
  if(dead){
    logMsg('灵脉·不敌守卫，撤退疗伤');
    toast('不敌守卫，已撤退');
    if(f.realm === 'lmg') S.lmgHp = lmgAttrD('hp'); else { S.maxHp = frMaxHpSuppressed(dmul(BASE_HP, dpow(BASE_GROW, S.level - 1))); S.hp = S.maxHp; }
    S.inMine = null; applyMineUI(); renderMai(); render(); save();
    return null;
  }
  /* 胜：占领 */
  if(f.hp <= 0){
    mn.occ = true; mn.st = Date.now(); mn.acc = 0;
    S.inMine = null;
    logMsg('灵脉·占领' + (f.realm === 'lmg' ? '灵矿' : '凡矿') + '（' + MAI_Q_N[mn.q] + '品 Lv.' + mn.lv + '）');
    toast('占领成功！' + MAI_Q_N[mn.q] + '品 Lv.' + mn.lv);
    applyMineUI(); renderMai(); render(); save();
    return null;
  }
  renderMineBar();
  return dmg;
}

/* 试算：在「不真实扣资源」的前提下算出本档位实际能升几级、共花多少 */
function maiPlan(costAt, realm, curLv, want, capLv){
  const m = maiData();
  const rk = realm === 'lmg' ? 'ly' : 'ore';
  let avail = m[rk], sum = 0, n = 0, lv = curLv;
  while(n < want){
    if(capLv !== null && lv >= capLv) break;
    const c = costAt(realm, lv);
    if(sum + c > avail) break;
    sum += c; lv++; n++;
  }
  return {n: n, cost: sum};
}

/* 2.00：批量升级试算（只算不扣，供按钮显示「本次能升几级 · 总价多少」） */
function farmUpPlan(want){
  const cap = farmLvCap();
  let lv = S.farmLv || 0, gem = resD('gem'), cnt = 0, cost = D(0);
  const w = Math.max(1, Math.floor(want) || 1);
  while(lv < cap && cnt < w){
    const c = Math.min(SAFE_INT, Math.round(25 * Math.pow(1.12, Math.max(0, lv - 1)) * Math.pow(1.5, S.farmStage || 0)));
    if(dcmp(gem, D(c)) < 0) break;
    gem = dsub(gem, D(c)); cost = dadd(cost, D(c)); lv++; cnt++;
  }
  return {cnt:cnt, cost:cost, lv:lv};
}
function farmUpgradeN(n){
  farmInit();
  if((S.farmLv || 0) >= farmLvCap()){ toast('已达当前阶上限 Lv.' + farmLvCap() + '，需先进阶'); return; }
  const want = (n === 0 || n === 'max') ? UI_BATCH_MAX : Math.max(1, Math.floor(n) || 1);
  let cnt = 0, lastCost = 0;
  while((S.farmLv || 0) < farmLvCap() && cnt < want){
    const c = farmUpCost();
    if(!resHas('gem', c)){ lastCost = c; break; }
    resPay('gem', c); S.farmLv++; cnt++;
  }
  if(cnt === 0){
    const short = dsub(D(lastCost || 0), resD('gem'));
    toast(dcmp(short, D(0)) > 0 ? ('宝石不足（还需 ' + dStr(short) + '）') : '当前无法继续升级');
    return;
  }
  toast('荒田 +' + cnt + ' 级 → Lv.' + S.farmLv + '（产量 +' + Math.round((farmGemMul() - 1) * 100) + '%）');
  logMsg('药园升级：荒田 +' + cnt + ' 级 → Lv.' + S.farmLv);
  renderFarm(); render(); save();
}

function relicLevelUpN(k, n){
  if(n === 0 || n === 'max'){ relicLevelUpMax(k); return; }

  if(lv0 >= RELIC_LV_MAX){ toast('已升满 ' + RELIC_LV_MAX + ' 级，请点击「突破」'); return; }

  for(let i = 0; i < n && target < RELIC_LV_MAX; i++){
    const need = relicFgNeed(target + 1, ad, k);
    const nx = dadd(acc, need);
    if(dcmp(nx, resD('relicFrag')) > 0) break;

  const r = RELICS.find(function(x){ return x.k === k; });

  }

  body.innerHTML = h;
}


const SUF = (function(){                     /* 生成至 1e306，即常规显示区 */
  const B = ['','K','M','B','T','Qa','Qi','Sx','Sp','Oc','No','Dc'];
  const U = ['','U','D','T','Qa','Qi','Sx','Sp','Oc','No'];
  const T = ['d','Vg','Tg','Qg','Qig','Sxg','Spg','Ocg','Nog','Cg'];
  for(let g = 12; g <= 102; g++){
    const k = g - 11;
    B.push(U[k % 10] + (T[Math.floor(k / 10)] || ''));
  }
  return B;
})();
function dStr(x){
  x = _toD(x);
  if(x.m === 0) return '0';
  if(!isFinite(x.m)) return '∞';
  const t = x.e;                                          /* BigInt */
  const tn = Number(t);
  if(tn < 0) return '0';
  if(tn < 6) return Math.round(x.m * Math.pow(10, tn)).toString();
  if(!S.sci && t <= 309n){                                /* 常规单位区（科学计数法开启时跳过） */
    const g = Math.floor(tn / 3);
    if(g < SUF.length) return (x.m * Math.pow(10, tn - g * 3)).toFixed(2) + SUF[g];
  }
  if(t < 1000000n) return x.m.toFixed(2) + 'e' + t.toString();   /* 10^310 ~ 10^999999 */
  const hd = Math.abs(x.m - 1) < 5e-3 ? '' : x.m.toFixed(2) + '\u00d7';
  return hd + '10^' + bigEStr(t);                         /* 再往上：2.00×10^(1.23e40) */
}
function fmtNum(x){ return (x && typeof x === 'object' && x.m !== undefined) ? dStr(x) : nStr(x); }
function nStr(n){
  if(n && typeof n === 'object' && n.m !== undefined) return dStr(n);   // 大数对象交给 dStr（科学/单位计数）
  if(typeof n !== 'number' || Number.isNaN(n)) return '0';    /* NaN 兜底（原误显示为 ∞） */
  if(!isFinite(n) || n > 1.7976931348623157e308) return '∞';
  if(n >= 1e6){
    if(S.sci) return n.toExponential(2);
    const g = Math.floor(Math.floor(Math.log10(n)) / 3);
    return g < SUF.length ? (n / Math.pow(10, 3 * g)).toFixed(2) + SUF[g] : n.toExponential(2); }
  if(Number.isInteger(n)) return n.toString();
  if(Math.abs(n) >= 100) return (Math.round(n * 10) / 10).toString();
  if(Math.abs(n) >= 1)  return (Math.round(n * 100) / 100).toString();   /* 1.05 → 1.05（原被舍成 1） */
  return (Math.round(n * 1000) / 1000).toString();                       /* 0.123 → 0.123 */
}

/* BigInt 存档序列化：JSON 不认 BigInt，用 \u0001 前缀标记后还原 */
function bigRepl(k, v){ return typeof v === 'bigint' ? '\u0001' + v.toString() : v; }
function bigRev(k, v){
  return (typeof v === 'string' && v.charCodeAt(0) === 1) ? BigInt(v.slice(1)) : v;
}

   凡人卷1.34 灵界数值全线性化：
     原设计 atk/hp 为指数成长（100×1.2^lv、500×1.15^lv），怪物血量/攻击也是指数
     （4200×1.285^层、42×1.235^层），升级成本更是 50×1.35^lv —— 指数叠指数，
     数值膨胀到 1e33 以上（修炼 Lv.400 时 atk 已达 5.45e33），且「投入指数上升、收益相对变薄」。

       · 妖魔血量 4200 + 6000×层数　妖魔攻击 42 + 25×层数
       · 升级成本 50 + 40×等级（原为指数，与线性收益不同阶会导致后期卡死）
     为什么线性体系自洽：击杀产出 ∝ 关卡（本就是线性）→ 累计灵力 ∝ t²，
     成本线性 → 等级 ∝ t，于是 atk ∝ t、怪物 HP ∝ t，比值收敛到常数。

     推关可持续 —— 既不卡死也不爆炸，数值最终停在 1e8 量级而非 1e33。

     全部仍能推关，不会卡死。
     必带的兼容性修复：atk/hp 类型由 exp 改为 lin 后，petPasSum() 与 runeSum() 里
     按 type==='exp' 判定会失效 —— 已镶符文与灵兽被动的攻击加成会凭空归零，两处均已补上 lin。
