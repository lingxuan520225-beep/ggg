/* ============ 属性加点 ============ */
function buy(k, delta){
  const a = ATTRS.find(x => x.k === k);
  const cur = S[k + 'Lv'];
  if(delta > 0){
    let bought = 0, paid = D(0);
    const room = isFinite(a.max) ? Math.max(0, a.max - cur) : Infinity;
    /* 上限文案：a.max 是「等级上限」（spd / combo 为 100 级），不是百分比。
       原把两者混为一谈、统一写成百分比，给攻速 / 连击率弹出「已达上限（百分比）」，语义错误。 */
    if(room <= 0){ toast(a.name + '已达上限 Lv.' + a.max); render(); return; }
    if(delta > 50000){                                   /* 大批量：闭式二分，一次算完 */
      bought = maxAfford(a, cur, Math.min(delta, isFinite(room) ? room : 1e15));
      paid = costFor(a, cur, bought);
      if(bought === 0) toast('转生币不足');
    }else{
      /* 与大批量的闭式结果对齐：先按闭式算出在预算内可买的最大级数，
         原逐次累加因「小值取整」规则不同，与闭式在相近次数上会有微小差额。 */
      const _cap = Math.min(delta, isFinite(room) ? room : 1e15);
      bought = maxAfford(a, cur, _cap);
      paid = bought > 0 ? costFor(a, cur, bought) : D(0);
      if(bought === 0){
        const c = aCost(a, cur);
        toast(dcmp(c, S.pts || D(0)) > 0 ? ('转生币不足（还需 ' + dStr(c) + '）') : (a.name + '已达上限 Lv.' + a.max));
      }
    }
    if(bought > 0){
      S[k + 'Lv'] += bought;
      S.pts = dadd(S.pts || D(0), dNeg(paid));
      if(dcmp(S.pts || D(0), D(0)) < 0) S.pts = D(0);     /* 闭式与逐次取整的微小误差兜底 */
      if(k === 'spd') startAuto();
    }
  }else{
    if(cur <= 0){ toast('当前等级为 0（可用「重置」清零）'); return; }
    const refund = aCost(a, cur - 1);
    S[k + 'Lv'] = cur - 1; S.pts = dadd(S.pts || D(0), refund);
    if(k === 'spd') startAuto();
    toast('已降级，返还 ' + dStr(refund) + ' 转生币');
  }
  render(); save();                                      /* 加点/降级立即落盘（原遗漏，与 resetOne 不一致） */
}

/* 连买 n 级总价（闭式）：Σ base*(1+0.5i), i = cur .. cur+n-1 */
function costFor(a, cur, n){
  if(!(n > 0)) return D(0);
  const N = D(n);
  return dadd(dmul(D(a.base), N),
    dmul(D(0.25 * a.base), dmul(N, dadd(dmul(D(2), D(cur)), dadd(N, D(-1))))));
}
/* 二分求在 cap 内最多能买多少级（大点数时避免百万次循环） */
function maxAfford(a, cur, cap){
  const pts = S.pts || D(0);
  if(!isFinite(cap) || !(cap > 0) || dcmp(costFor(a, cur, 1), pts) > 0) return 0;
  let lo = 1, hi = Math.min(Math.floor(cap), 1e12);      /* 防御 Infinity：调用处已兜底，此处再加一层保险 */
  if(dcmp(costFor(a, cur, hi), pts) <= 0) return hi;
  while(lo < hi){
    const mid = Math.floor((lo + hi + 1) / 2);
    if(dcmp(costFor(a, cur, mid), pts) <= 0) lo = mid; else hi = mid - 1;
  }
  return lo;
}
function refundFor(a, lv){
  if(lv > 200000){                                  /* 大等级闭式：Σ base*(1+0.5i) */
    const n = D(lv);
    return dmul(D(a.base), dadd(n, dmul(D(0.25), dmul(n, D(lv - 1)))));
  }
  let r = D(0);
  for(let i = 0; i < lv; i++) r = dadd(r, aCost(a, i));
  return r;
}
function resetOne(k){
  const a = ATTRS.find(x => x.k === k);
  if(!a) return;
  const lv = S[k + 'Lv'];
  if(!lv){ toast(a.name + '当前等级为 0'); return; }
  const refund = refundFor(a, lv);
  S[k + 'Lv'] = 0;
  S.pts = dadd(S.pts || D(0), refund);
  if(k === 'spd') startAuto();
  toast(a.name + '已重置，返还 ' + dStr(refund) + ' 转生币');
  render(); save();
}
function resetAttrs(){
  let refund = D(0);
  for(const a of ATTRS){
    refund = dadd(refund, refundFor(a, S[a.k + 'Lv']));
    S[a.k + 'Lv'] = 0;
  }
  S.pts = dadd(S.pts || D(0), refund);
  startAuto();
  toast('属性已重置，返还 ' + dStr(refund) + ' 转生币');
  render(); save();                                      /* 一键重置立即落盘（原遗漏） */
}

/* 存档下载 / 上传：每天各 3 次（用于防止频繁倒腾存档） */
const SAVE_DL_LIMIT = 3, SAVE_UL_LIMIT = 3;
function dlLeft(){
  const d = today();
  if(S.expDay !== d){ S.expDay = d; S.expCnt = 0; }
  return Math.max(0, SAVE_DL_LIMIT - (S.expCnt || 0));
}
function ulLeft(){
  const d = today();
  if(S.impDay !== d){ S.impDay = d; S.impCnt = 0; }
  return Math.max(0, SAVE_UL_LIMIT - (S.impCnt || 0));
}
function exportSave(){
  if(dlLeft() <= 0){ toast('今日下载次数已用完（每天 ' + SAVE_DL_LIMIT + ' 次）'); return; }
  const d0 = today();
  if(S.expDay !== d0){ S.expDay = d0; S.expCnt = 0; }
  S.expCnt = (S.expCnt || 0) + 1;
  save();                                   /* 立即落盘：原只改内存，导出后立刻刷新/关页会丢失计数，等于绕过每日 3 次 */
  try{
    const data = localStorage.getItem('xiantu_save') || '{}';
    const code = 'XT1.' + btoa(unescape(encodeURIComponent(data)));
    const ta = $('setTa'); ta.value = code; ta.focus(); ta.select();
    if(navigator.clipboard && navigator.clipboard.writeText){
      navigator.clipboard.writeText(code).then(function(){ toast('存档码已复制到剪贴板'); }, function(){ toast('请长按文本框全选复制'); });
    } else { toast('请长按文本框全选复制'); }
  }catch(e){ toast('导出失败'); }
}
function importSave(){
  const v = ($('setTa').value || '').trim();
  if(!v){ toast('请先粘贴存档码'); return; }
  /* 长度上限：粘贴超长文本会让 atob / JSON.parse 长时间阻塞主线程，页面直接卡死 */
  if(v.length > SAVE_TA_MAX){ toast('存档码过长，请确认粘贴内容'); return; }
  if(ulLeft() <= 0){ toast('今日上传次数已用完（每天 ' + SAVE_UL_LIMIT + ' 次）'); return; }
  const d0 = today();
  if(S.impDay !== d0){ S.impDay = d0; S.impCnt = 0; }
  S.impCnt = (S.impCnt || 0) + 1;
  try{
    let raw = v.indexOf('XT1.') === 0 ? decodeURIComponent(escape(atob(v.slice(4)))) : v;
    const o = JSON.parse(raw);
    if(typeof o !== 'object' || o === null || o.level === undefined) throw 0;
    /* 次数以本地为准写入被导入档：原直接整包覆盖，导入旧档即可把次数刷回 0，等于绕过限制 */
    o.impDay = S.impDay; o.impCnt = S.impCnt;
    o.expDay = S.expDay; o.expCnt = S.expCnt || 0;
    /* 改写次数后必须重新签名并同步备份快照：
       签名覆盖存档全部字段（含 _bak 自身）。若只改值不重签，load() 会判定「签名不符 = 篡改」，
       随后用 _bak 把次数回滚成「导出时的旧值」 —— 结果是导入不计次，可反复导入绕过每日 3 次限制
       （1.22 声称修过，但被签名回滚抵消，实际一直可绕过）。 */
    try{
      if(o._bak && typeof o._bak === 'object'){
        ['impDay','impCnt','expDay','expCnt'].forEach(function(k){ o._bak[k] = o[k]; });
      }
      o._sig = sigOf(o);
    }catch(_e){}
    localStorage.setItem('xiantu_save', JSON.stringify(o, bigRepl));
    toast('导入成功，即将刷新'); setTimeout(function(){ location.reload(); }, 800);
  }catch(e){ toast('存档码无效'); }
}

   高层奖励相对血量几乎可以忽略，爬塔收益完全不划算。
   现叠加指数项 1.025^(f-1)：第 100 层约 ×11、第 300 层约 ×1400、第 1000 层约 ×1.6e10。
   用 dpow（对数法，O(1)）直接算，层数极大也不会溢出为 Infinity。 */
/* ---- 1.56 双塔掉落加成（两塔 keys 完全分开，互不通用）----
   凡人塔 twpts：转生点百分比；神兵塔 sbpts：碎片数量百分比。
   目标总倍率 ≈ 50 倍（不含香火·塔神，香火另计），分配：
     成就 300% + 宝物 2300%（800/1500）+ 功法 1300%（400/900）+ 称号 1000% = 4900% ≈ 50x */
function frTowerPct(){                                  /* 凡人塔转生点加成 %（含成就 + 香火·塔神） */
  return achFrPts() + treSum('twpts') + gfSum('twpts') + titleSum('twpts');
}
function achSbPts(){ return 30 * achSt('sbF') + incensePct('tower'); }   /* 神兵塔：香火·塔神同样生效 */
function sbTowerPct(){                                  /* 神兵塔碎片加成 %（含成就 + 香火·塔神） */
  return achSbPts() + treSum('sbpts') + gfSum('sbpts') + titleSum('sbpts');
}
function towerPtsFor(f){
  if(!(f > 0)) return D(0);
  const poly = D(10*f + f*f/2);
  const grow = (f > 1) ? dpow(D(1.025), f - 1) : D(1);
  return dmul(dmul(poly, grow), D(1 + frTowerPct()/100));
}
/* 神兵塔碎片加强：原 1 + f/5（第 100 层仅 21 枚），现追加二次项 f²/100，
   第 100 层 121 枚、第 300 层 961 枚、第 500 层 2601 枚。 */
function sbDropNum(f){

  return Math.floor(base * (1 + sbTowerPct()/100)) + achSbDrop();   /* 百分比先乘后取整，整数特殊奖励最后加 */
}

/* resetToMain() 名为「回到主线」，实际还负责：清毒层、清不死不休层数、重启 autoTimer。
   耦合偏重 —— 新增模式（如灵界塔）时必须在此同步加分支，否则会重演「秘境中转生后攻击永久停止」。 */
function enterTower(t){
  t = t || 'fr';
  exitLingToFr();                     /* 灵界与凡人界互斥：进塔前先返回凡人界，否则攻击会打到灵界怪物 */
  cancelSecret();
  if(S.inSecret){ exitSecret(); }
  clearBossMode();                    /* 挑战无限塔时自动退出 BOSS 挑战（内部已清理标记并刷新 UI） */
  S.towerType = t;
  /* 1.55：起始层 = 已通关层 + 1 + 该塔专属的跳跃宝物加成（凡人塔 twfr / 神兵塔 twsb，互不通用） */
  const _skip = treSum(t === 'sb' ? 'twsb' : 'twfr');
  S.towerFloor = (t==='sb' ? ((S.sbDone||0)+1) : ((S.towerDone||0)+1)) + _skip;
  S.towerMaxHp = towerMaxHpFor(S.towerFloor, t);
  S.towerHp = S.towerMaxHp;
  S.inTower = true;
  S.poison = 0;
  applyTowerUI();
  renderTowerMode();
  logMsg('挑战'+towerName(t)+'第 '+S.towerFloor+' 层！');
  toast(towerName(t)+' 第 '+S.towerFloor+' 层');
  save();
}
function exitTower(){
  S.inTower=false; S.towerType='fr';
  clearBossMode();                    /* 返回主线时自动退出 BOSS 挑战 */
  applyTowerUI();
  applySecretUI();
  render();
  logMsg('离开塔，返回主线');
  toast('返回主线');
  save();
}
function doTowerHit(manual){
  hitInc();
  const ty = S.towerType || 'fr';
  if(!S.towerHp || dcmp(S.towerHp, D(0)) <= 0){ S.towerHp = S.towerMaxHp || towerMaxHpFor(S.towerFloor||1, ty); }
    const _atk = calcAtk();                                /* 缓存攻击力（原每连击段重复计算，段数多时开销大） */
  let sum = dmul(_atk, D(0.9 + Math.random() * 0.2));
  let critN = 0, comboN = 0;
  const crit = Math.random() * 100 < calcCrit();
  if(crit){ sum = dmul(sum, D(critMult())); critN = 1 + critExtra();
    }
  sum = dadd(sum, dmul(_atk, D(calcIgn() / 100)));        /* 无视防御附加真伤（原三模式完全缺失） */
  { const _cs = {n:0, hit:false};
    sum = comboAdd(sum, _atk, 0.15, 0.9, 0.2, _cs);
    if(_cs.hit) comboN += _cs.n; }
  { const _pn = gfSum('psnhit');
    if(_pn > 0 && S.poison > 0) sum = dadd(sum, dmul(dmul(D(S.poison), D(_pn)), D(psnMultSum()))); }
  sum = dmul(sum, D(1 + tSum('nmult') / 100));            /* 猛攻之势 / 屠戮本能 */
  sum = dmul(sum, D(1 + calcFdm() / 100));                /* 最终伤害乘区 */
  if(ignAmpMul() > 1) sum = dmul(sum, D(ignAmpMul()));    /* 破界神锋 */
  S.ndxStack += 1 + tSum('ndxplus');                      /* 不死不休叠层（原三模式完全不叠） */
  sum = dmul(sum, D(1 + calcNdx() / 100));                /* 不死不休伤害加成 */
if(manual) S.clicks++;
  S.poison = Math.min(poisonCap(), S.poison + poisonAdd());   /* 自动/手动都叠毒层 */
  S.totalDmg = dadd(S.totalDmg || D(0), sum);
  S.towerHp = dadd(S.towerHp, dNeg(sum));
  if(dcmp(S.towerHp, D(0)) < 0) S.towerHp = D(0);
  let poison = D(0);
  if(psnPerStack() > 0){ poison = calcPsnDmg(S.poison);
    S.totalDmg = dadd(S.totalDmg || D(0), poison);
    S.towerHp = dadd(S.towerHp, dNeg(poison));
    if(dcmp(S.towerHp, D(0)) < 0) S.towerHp = D(0);
  }
  achHit(sum, crit, poison);                           /* 移出毒伤分支 */
  /* 斩杀线天赋（原仅主线判定，塔中失效） */
  const _line = tSum('line');
  if(_line > 0 && dcmp(S.towerHp, dmul(S.towerMaxHp, D(_line / 100))) <= 0 && dcmp(S.towerHp, D(0)) > 0) S.towerHp = D(0);
  if(dcmp(S.towerHp, D(0)) <= 0){ towerKill(); }
  return {sum:sum, crit:crit, critN:critN, combo:comboN>0, comboN:comboN, poison:poison};
}
function towerKill(){
  S.ndxStack = 0;                                    /* 击杀后清空不死不休层数（主线 killBoss 已清，此处补齐） */
  resAdd('soul', soulGain());                            /* 灵魂：击杀 +1 */
  const f = S.towerFloor;
  if(S.towerType === 'sb'){
    const d = sbDrop(f);
    S[d.key] = (S[d.key]||0) + d.n;
    questAdd('kill', 1);
    { const _bn = frBoxDrop(BOX_DROP_TOWER); if(_bn > 0){ resAdd('box', _bn); questAdd('getbox', _bn); logMsg('神兵塔第 '+f+' 层掉落凡尘宝箱 ×'+_bn); } }
    if(f > (S.sbDone||0)) S.sbDone = f;
    if(f > (S.stSbF||0)) S.stSbF = f;
    S.poison = 0;                                    /* 每层换怪，毒层从 0 重新叠 */
    logMsg('神兵塔第 '+f+' 层通关！掉落'+d.nm+'碎片 ×'+d.n);
    toast('第 '+f+' 层！'+d.nm+'碎片 +'+d.n);
    /* 1.56：跃层宝物【每次击败都生效】——原仅进入塔时跳一次，买了等于只有一次收益。
       现每层通关都 +treSum('twsb')，满四件每次跳 100 层。
       起点仍受 towerFloor 影响，故用 S.sbDone 重算，避免连续跳层时层数被上一次的起点带偏。 */
    S.towerFloor = f + 1 + treSum('twsb');
    S.towerMaxHp = towerMaxHpFor(S.towerFloor,'sb');
    S.towerHp = S.towerMaxHp;
    renderTowerMode(); save(); return;
  }
  const gain = towerPtsFor(f);
  S.pts = dadd(S.pts || D(0), gain);                 /* 转生点奖励（已是大数，无需再 D() 包一层） */
  questAdd('kill', 1);
  { const _bn = frBoxDrop(BOX_DROP_TOWER); if(_bn > 0){ resAdd('box', _bn); questAdd('getbox', _bn); logMsg('凡人塔第 '+f+' 层掉落凡尘宝箱 ×'+_bn); } }
  if(f > (S.towerDone||0)) S.towerDone = f;
  if(f > (S.stFrF||0)) S.stFrF = f;
  S.poison = 0;                                     /* 每层换怪，毒层从 0 重新叠 */
  logMsg('凡人塔第 '+f+' 层通关！转生点 +'+dStr(gain));
  toast('第 '+f+' 层通关！转生点 +'+dStr(gain));
  /* 凡人塔改为无限层：原满 1000 层会强制 exitTower() 退出并锁死入口，
     进度停在 1000 不再增长；现直接推进到下一层，可无限爬。
     1.56：跃层宝物【每次击败都生效】——原仅进入塔时跳一次，买了等于只有一次收益。
     现每层通关都 +treSum('twfr')，满四件每次跳 100 层。 */
  S.towerFloor = S.towerDone + 1 + treSum('twfr');
  S.towerMaxHp = towerMaxHpFor(S.towerFloor,'fr');
  S.towerHp = S.towerMaxHp;
  renderTowerMode(); save();
}
/* 敌人区视觉统一同步：wolfImg / towerImg / towerImgSB 由本函数独家控制。
   原三个 apply*UI() 都直接操作 wolfImg，靠「谁最后调用谁赢」的隐式约定 ——
   applyBossUI() 是收尾调用且无条件 display=on?'block':'none'，退出 BOSS 时会
   把秘境 / 塔的立绘一并抹掉（初始化也必须严格按 secret→tower→boss 顺序，非常脆弱）。
   现三个 apply*UI() 只管各自的状态标记与 modeBar，本函数按优先级决定谁可见，
   调用顺序不再有任何影响。 */
function syncEnemyVisual(){
  /* 底层主线怪图（.boss-img）由 applyRealmUI 在进入灵界时隐藏；
     本函数是立绘显示的唯一真相源，凡人界所有路径都必须先把它恢复，
     否则「进过一次灵界后主线怪物图永久不显示」。 */
  document.querySelectorAll('.boss-img').forEach(function(e){ e.style.display = ''; });
  const wl = $('wolfImg'), wt = $('towerImg'), ws = $('towerImgSB');
  if(S.inBoss){                                   /* 优先级最高：BOSS 挑战立绘 */
    if(ws) ws.style.display = 'none';
    if(wt) wt.style.display = 'none';
    if(wl){ wl.style.display = 'block'; setEnemyImg('boss', S.bossIdx); }
    return;
  }
  if(S.inSecret){                                 /* 秘境：覆盖层显示首领/主线图 */
    if(ws) ws.style.display = 'none';
    if(wt) wt.style.display = 'none';
    if(wl){ wl.style.display = 'block'; setEnemyImg('secret'); }
    return;
  }
  if(S.inTower){                                  /* 塔：用专属图，覆盖层让位 */
    if(wl) wl.style.display = 'none';
    if(wt) wt.style.display = (S.towerType === 'sb') ? 'none' : 'block';
    if(ws) ws.style.display = (S.towerType === 'sb') ? 'block' : 'none';
    return;
  }
  /* 主线：隐藏全部覆盖层，露出底层 .boss-img（主线怪图） */
  if(wl) wl.style.display = 'none';
  if(wt) wt.style.display = 'none';
  if(ws) ws.style.display = 'none';
  setEnemyImg('main');
}
function applyTowerUI(){
  const on = !!S.inTower, sb = on && S.towerType==='sb';
  const bar = $('towerModeBar'); if(bar) bar.style.display = on ? '' : 'none';
  syncEnemyVisual();                              /* 只设置本模式状态与 modeBar，图片统一由 syncEnemyVisual 决定 */
}
function renderTowerMode(){
  if(!S.inTower) return;
  $('tmName').textContent = towerName();
  $('tmFloor').textContent = S.towerFloor;
  const pw = $('tmPtsWrap'); if(pw) pw.style.display = (S.towerType==='sb') ? 'none' : '';
  if(pw && S.towerType!=='sb') $('tmPts').textContent = dStr(S.pts || D(0));
  if(S.towerMaxHp && S.towerHp){
    const dh = dcmp(S.towerHp, D(0)) < 0 ? D(0) : S.towerHp;
    const ratio = dNum(ddiv(dh, S.towerMaxHp));                 /* 统一走 dRatio/dNum，避免手写指数运算在量级悬殊时下溢为 0 */
    const pct = Math.max(0, Math.min(100, ratio*100));
    $('hpFill').style.width = pct+'%';
    $('hpText').textContent = dStr(dh)+' / '+dStr(S.towerMaxHp);
  }
}

function bossCntArr(k){
  if(!S.bossCnt) S.bossCnt = {};
  if(!S.bossCnt[k]) S.bossCnt[k] = BOSS_LIST.map(function(){ return 0; });   /* 长度跟随 BOSS_LIST（现 9 个） */
  while(S.bossCnt[k].length < BOSS_LIST.length) S.bossCnt[k].push(0);
  return S.bossCnt[k];
}
function bossBaseHp(i){ return dmul(D(1000000), dpow(D(3), i)); }
/* 界别只有唯一真相源：S.bossWorld。
   原另有模块变量 bossW，两者语义相同却要在 openBoss / switchBossWorld / applyBossUI / load 四处同步，
   任何一条路径漏同步就会把通关进度写进错误的界别数组（读档恢复时 bossW 仍是默认 'fj' 即典型）。
   switchBossWorld 里那句「bossW = k; S.bossWorld = k;」的双写本身就是冗余信号——既然总是同时改，
   说明设计意图上就是同一个东西。现统一为 bossWorld() 读取函数。 */
function bossWorld(){ const k = S.bossWorld || 'fj'; return BOSS_WORLDS.some(x => x.k === k) ? k : 'fj'; }
/* 第 i 关基础血量 ×3^i（原 ×30 → ×6 → ×3，第1关不变） */

function bossUnlocked(i){ if(i === 0) return true; return (bossCntArr(bossWorld())[i-1] || 0) > 0; }

function renderBoss(){
  if(!$('bossGrid')) return;
  $('bossWinTotal').textContent = nStr(S.bossWin || 0);
  const arr = bossCntArr(bossWorld());
  let html = '';
  for(let i = 0; i < BOSS_LIST.length; i++){
    const b = BOSS_LIST[i], rw = BOSS_REWARDS[b.t], cnt = arr[i] || 0;
    const un = bossUnlocked(i);
    const cur = S.inBoss && S.bossWorld === bossWorld() && S.bossIdx === i;
    html += '<div class="bs-cell' + (un ? '' : ' lock') + (cur ? ' cur' : '') + '">';
    html +=   '<div class="bs-head"><span class="bs-name">' + (i+1) + '. ' + b.n + '</span>';
    html +=     '<span class="bs-cnt">通关 ' + nStr(cnt) + ' 次</span></div>';
    html +=   '<div class="bs-hp">血量 ' + dStr(bossCurMaxHp(i)) + '</div>';
    html +=   '<div class="bs-rw">奖励：<b>' + rw.n + ' +' + nStr(rw.val) + rw.unit + '</b>/次</div>';
    html +=   '<div class="bs-have">已获 ' + nStr((S['boss_'+b.t] || 0) * rw.val) + rw.unit
            + '（' + nStr(S['boss_'+b.t] || 0) + ' 层）</div>';
    html +=   '<button class="bs-go' + (un ? '' : ' dis') + '"' + (un ? ' onclick="enterBoss(' + i + ')"' : ' disabled')
            + '>' + (cur ? '挑战中' : (un ? '挑 战' : '需通关前一关')) + '</button>';
    html += '</div>';
  }
  $('bossGrid').innerHTML = html;
}
function enterBoss(i){
  exitLingToFr();                     /* 同上：挑战 BOSS 前先返回凡人界 */
  /* 越界兜底：与 renderBossMode / bossKill 一致。存档被篡改时 bossCntArr 可能比 BOSS_LIST 长，
     此时 bossUnlocked(i) 会误判为真，下面 BOSS_LIST[i].n 直接抛 TypeError 带崩战斗循环。 */
  i = i | 0;
  if(!(i >= 0) || i >= BOSS_LIST.length){ toast('BOSS 数据异常'); return; }
  if(!bossUnlocked(i)){ toast('需先通关前一关'); return; }
  /* 从秘境 / 塔 进入挑战BOSS：先退出原模式，否则 render() 会优先渲染原模式导致「点了没切换」 */
  /* 注：exitSecret() 内部会调用 clearBossMode()，在此路径下属冗余调用（此时 S.inBoss 必为 false），
     但 clearBossMode 有 if(!S.inBoss) return 兜底，无害；保留以兼容「秘境中残留 BOSS 状态」的极端情况。 */
  if(S.inSecret){ exitSecret(); }
  if(S.inTower){ exitTower(); }
  S.inBoss = true; S.bossWorld = bossWorld(); S.bossIdx = i;
  S.bossMaxHp = bossCurMaxHp(i); S.bossHp = S.bossMaxHp;
  S.poison = 0;
  applyBossUI(); render();                             /* render 内 inBoss 分支会调用 renderBossMode 刷新血条 */
  closeBoss();
  logMsg('挑战【' + BOSS_LIST[i].n + '】！第 ' + ((bossCntArr(bossWorld())[i] || 0) + 1) + ' 次');
  toast('BOSS 挑战开始');
  save();
}

  if(!S.inBoss) return;
  S.inBoss = false;
  applyBossUI();
}
function exitBoss(){
  S.inBoss = false; applyBossUI(); render();
  logMsg('退出 BOSS 挑战，返回主线'); toast('返回主线'); save();
}
function applyBossUI(){
  const on = !!S.inBoss;
  const bar = $('bossModeBar'); if(bar) bar.style.display = on ? '' : 'none';
  syncEnemyVisual();                     /* 同上：本函数不直接操作图片，避免「最后调用者胜出」 */
}
function renderBossMode(){
  if(!S.inBoss) return;
  /* bossIdx 越界防御：旧档字段缺失或 BOSS_LIST 增删后可能为 undefined/负数/越界，
     BOSS_LIST[undefined].n 会抛 TypeError，render() 整体崩溃、界面彻底卡死。
     越界时直接静默退出 BOSS 模式并收尾，而不是让整个渲染挂掉。 */
  const _bi = (typeof S.bossIdx === 'number' && isFinite(S.bossIdx)) ? (S.bossIdx | 0) : 0;
  const _bd = BOSS_LIST[_bi];
  if(!_bd){ S.inBoss = false; if(typeof applyBossUI === 'function') applyBossUI(); return; }
  if(S.bossIdx !== _bi) S.bossIdx = _bi;                    /* 归一化非法/小数下标，避免后续写入错位 */
  $('bmName').textContent = _bd.n;
  $('bmCnt').textContent = (bossCntArr(bossWorld())[_bi] || 0) + 1;
  if(S.bossMaxHp && S.bossHp){
    const dh = dcmp(S.bossHp, D(0)) < 0 ? D(0) : S.bossHp;
    const ratio = dNum(ddiv(dh, S.bossMaxHp));                 /* 同上 */
    const pct = Math.max(0, Math.min(100, ratio * 100));
    $('hpFill').style.width = pct + '%';
    $('hpText').textContent = dStr(dh) + ' / ' + dStr(S.bossMaxHp);
  }
}
function doBossHit(manual){
  hitInc();
  if(!S.bossHp || dcmp(S.bossHp, D(0)) <= 0){ S.bossMaxHp = bossCurMaxHp(S.bossIdx); S.bossHp = S.bossMaxHp; }
    const _atk = calcAtk();                                /* 缓存攻击力（原每连击段重复计算，段数多时开销大） */
  let sum = dmul(_atk, D(0.92 + Math.random() * 0.16));
  let critN = 0, comboN = 0;
  const crit = Math.random() * 100 < calcCrit();
  if(crit){ sum = dmul(sum, D(critMult())); critN = 1 + critExtra();
    }
  sum = dadd(sum, dmul(_atk, D(calcIgn() / 100)));        /* 无视防御附加真伤（原三模式完全缺失） */
  { const _cs = {n:0, hit:false};
    sum = comboAdd(sum, _atk, 0.15, 0.92, 0.16, _cs);
    if(_cs.hit) comboN += _cs.n; }
  { const _pn = gfSum('psnhit');
    if(_pn > 0 && S.poison > 0) sum = dadd(sum, dmul(dmul(D(S.poison), D(_pn)), D(psnMultSum()))); }
  sum = dmul(sum, D(1 + tSum('nmult') / 100));            /* 猛攻之势 / 屠戮本能 */
  sum = dmul(sum, D(1 + calcFdm() / 100));                /* 最终伤害乘区 */
  if(ignAmpMul() > 1) sum = dmul(sum, D(ignAmpMul()));    /* 破界神锋 */
  S.ndxStack += 1 + tSum('ndxplus');                      /* 不死不休叠层（原三模式完全不叠） */
  sum = dmul(sum, D(1 + calcNdx() / 100));                /* 不死不休伤害加成 */
if(manual) S.clicks++;
  S.poison = Math.min(poisonCap(), S.poison + poisonAdd());
  S.totalDmg = dadd(S.totalDmg || D(0), sum);
  S.bossHp = dadd(S.bossHp, dNeg(sum));
  if(dcmp(S.bossHp, D(0)) < 0) S.bossHp = D(0);
  let poison = D(0);
  if(psnPerStack() > 0){ poison = calcPsnDmg(S.poison);
    S.totalDmg = dadd(S.totalDmg || D(0), poison);
    S.bossHp = dadd(S.bossHp, dNeg(poison));
    if(dcmp(S.bossHp, D(0)) < 0) S.bossHp = D(0);
  }
  achHit(sum, crit, poison);                           /* 移出毒伤分支 */
  /* 斩杀线天赋（原仅主线判定，BOSS 中失效） */
  const _line = tSum('line');
  if(_line > 0 && dcmp(S.bossHp, dmul(S.bossMaxHp, D(_line / 100))) <= 0 && dcmp(S.bossHp, D(0)) > 0) S.bossHp = D(0);
  if(dcmp(S.bossHp, D(0)) <= 0) bossKill();
  return {sum:sum, crit:crit, critN:critN, combo:comboN>0, comboN:comboN, poison:poison};
}
function bossKill(){
  S.ndxStack = 0;                                    /* 击杀后清空不死不休层数（主线 killBoss 已清，此处补齐） */
  resAdd('soul', soulGain());                            /* 灵魂：击杀 +1 */
  /* 越界防御：与 renderBossMode() 保持一致。存档被污染 / BOSS_LIST 增删条目 / 旧档缺字段时
     b 为 undefined，b.t 直接抛 TypeError，会把整个战斗循环带崩、界面卡死。
     这里静默退出 BOSS 模式，不发放奖励（奖励依赖 b.t，无法安全计算）。 */
  const i = S.bossIdx | 0;
  const b = BOSS_LIST[i];
  if(!b || !BOSS_REWARDS[b.t]){ S.inBoss = false; applyBossUI(); return; }
  const rw = BOSS_REWARDS[b.t];
  const arr = bossCntArr(bossWorld());
  /* 顺序不可调换：必须先递增通关计数，下面 bossCurMaxHp(i) 用的是「递增后」的次数
     （第 N 次挑战血量更高）。若把递增挪到后面，下一只会拿到旧血量，难度曲线失效。 */
  arr[i] = (arr[i] || 0) + 1;
  S['boss_' + b.t] = (S['boss_' + b.t] || 0) + 1;
  S.bossWin = (S.bossWin || 0) + 1;
  S.poison = 0;
  logMsg('通关【' + b.n + '】第 ' + arr[i] + ' 次！' + rw.n + ' +' + nStr(rw.val) + rw.unit);
  toast('通关！' + rw.n + ' +' + nStr(rw.val) + rw.unit);
  questAdd('kill', 1);
  S.bossMaxHp = bossCurMaxHp(i);          /* 下次挑战血量 ×2，无限挑战 */
  S.bossHp = S.bossMaxHp;
  renderBossMode(); render(); save();
  /* 面板开着时同步刷新：renderBossMode 只更新战斗条，通关次数 / 已获层数要等重开面板才变。
     renderBoss() 内部取 $('bossGrid')，面板未打开时自然空跑，代价可忽略。 */
  renderBoss();
}

  logMsg(w.n+' 提升至 Lv '+(lv+1));

}

const RELIC_BREAK_MAX = 100;                              /* 可突破次数上限 */
const RELIC_AD_COEF   = 1.0;                              /* 战斗类遗物：每次突破的效果增量（线性系数） */
/* 2.08 产出类遗物（c === 'src'：内力 / 凡尘宝箱 / 种子等资源向）突破增量单独压低到 0.25。
   前期（未突破）完全不受影响，只压后期高突破段。 */
const RELIC_AD_COEF_RES = 0.25;
function relicAdCoef(c){ return c === 'src' ? RELIC_AD_COEF_RES : RELIC_AD_COEF; }
const RELIC_AD_POW    = 2;                                /* 消耗的多项式次数（1 = 纯线性，2 = 平方） */
function relicAd(k){ return (S.relicAd && S.relicAd[k]) || 0; }          /* 突破次数 0~100 */
function relicLv(k){ return (S.relicLv && S.relicLv[k]) || 0; }          /* 当前升级等级 0~10 */
/* 遗物升级（1~10 级）所需碎片：随等级递增，并随突破次数 ×2.5^突破 */
/* 遗物消耗（大数）：原用 Math.round(k * Math.pow(r, ad))，ad 上限 100 ——
   2.5^100 ≈ 6e39、3^100 ≈ 5e47、2.4^100 ≈ 4e38，均远超 Number.MAX_SAFE_INTEGER(9.007e15)：
   · 尾数精度丢失，费用不再精确，同一次突破在 UI 与校验处可能算出不同值
   · Math.round 对 >2^53 的数无效（本来就不是整数），形同虚设
   改用大数幂运算，精确表示与比较；扣费时若超出安全整数则直接判「资源不可能够」。 */
function relicFgNeed(lv, ad, k){
  const m = RELIC_Q_MUL[Math.max(0, Math.min(RELIC_Q_MUL.length - 1, relicQ(k)))];
  return dmul(dmul(D(10 + 6 * ((lv || 1) - 1)), dpow(D(1 + (ad || 0)), RELIC_AD_POW)), D(m));
}
function relicBreakFrag(ad, k){
  const m = RELIC_Q_MUL[Math.max(0, Math.min(RELIC_Q_MUL.length - 1, relicQ(k)))];
  return dmul(dmul(D(200), dpow(D(1 + (ad || 0)), RELIC_AD_POW)), D(m));
}
function relicBreakGem(ad, k){
  const m = RELIC_Q_MUL[Math.max(0, Math.min(RELIC_Q_MUL.length - 1, relicQ(k)))];
  return dmul(dmul(D(5000), dpow(D(1 + (ad || 0)), RELIC_AD_POW)), D(m));
}
/* 大数费用扣减：费用在双精度安全整数内才按 Number 精确扣；超出则玩家不可能持有这么多，
   直接返回 false（避免 Number 减法丢失精度导致「扣了但没扣够」）。 */
function payCost(key, costD){
  const v = dNum(costD);
  if(!isFinite(v) || v > Number.MAX_SAFE_INTEGER) return false;
  const cur = S[key] || 0;
  if(cur < v) return false;
  S[key] = cur - v;
  return true;
}
/* 大数费用是否足够（避免 Number 直接比较时 1e40 与 Infinity/大数对象行为不一致） */
function canPay(key, costD){ return dcmp(D(S[key] || 0), costD) >= 0; }
/* 遗物效果 = 基础 × (1 + 20%×升级等级) × (1 + COEF×突破次数)
   2.07：原「2^突破次数」为指数，突破 29 次即 ×5.4e8，是后期数值失控的主因之一，改为线性。 */
function relicVal(k){
  if(!S.relics || !S.relics[k]) return 0;
  const r = RELICS.find(function(x){ return x.k === k; });
  if(!r) return 0;
  return Math.round(r.v * (1 + 0.20 * relicLv(k)) * (1 + relicAdCoef(r.c) * relicAd(k)));
}

function relicLevelUp(k){

  payCost('relicFrag', need);
  S.relicLv = S.relicLv || {}; S.relicLv[k] = lv + 1;
  const r = RELICS.find(function(x){ return x.k === k; });

  logMsg('【' + r.n + '】升级至 ' + S.relicLv[k] + '/' + RELIC_LV_MAX + ' 级，效果 ' + relicVal(k));

/* 1.56 遗物一键升级：把当前碎片能支撑的等级一次升满（封顶 RELIC_LV_MAX），
   避免后期每级都要点几百次「升级」。逐档试算累计消耗，够多少升多少。 */
function relicLevelUpMax(k){
  if(!S.relics || !S.relics[k]){ toast('尚未拥有此遗物'); return; }
  const lv0 = relicLv(k), ad = relicAd(k);
  if(lv0 >= RELIC_LV_MAX){ toast('已升满 ' + RELIC_LV_MAX + ' 级，请点击「突破」'); return; }
  let acc = D(0), target = lv0;
  for(let lv = lv0 + 1; lv <= RELIC_LV_MAX; lv++){
    const need = relicFgNeed(lv, ad, k);
    const next = dadd(acc, need);
    if(dcmp(next, resD('relicFrag')) > 0) break;      /* 碎片不够，停在上一档 */
    acc = next; target = lv;
  }
  if(target === lv0){
    const need = relicFgNeed(lv0 + 1, ad, k);
    toast('遗物碎片不足（需 ' + dStr(need) + '，当前 ' + dStr(resD('relicFrag')) + '）');
    return;
  }
  payCost('relicFrag', acc);
  S.relicLv = S.relicLv || {}; S.relicLv[k] = target;
  const r = RELICS.find(function(x){ return x.k === k; });
  toast('【' + r.n + '】一键升至 ' + target + ' / ' + RELIC_LV_MAX + ' 级（耗碎片 ' + dStr(acc) + '）');
  logMsg('【' + r.n + '】一键升级 ' + lv0 + ' → ' + target + ' 级，耗碎片 ' + dStr(acc) + '，效果 ' + relicVal(k));
  renderRelic(); render(); save();
}
/* 全部已拥有遗物一键升级（按「每级性价比」从便宜到贵依次升，尽量把碎片花在刀刃上） */
function relicLevelUpAll(){

                    .filter(function(r){ return relicLv(r.k) < RELIC_LV_MAX; });

  own.forEach(function(r){ const before = relicLv(r.k); relicLevelUpMax(r.k); if(relicLv(r.k) > before) n++; });
  if(n) toast('已升级 ' + n + ' 件遗物');
  renderRelic(); render(); save();
}
/* 遗物突破：需先升满 10 级；消耗大量碎片 + 大量宝石；突破后等级归零、效果翻倍 */
function relicAdvance(k){
  if(!S.relics || !S.relics[k]){ toast('尚未拥有此遗物'); return; }
  const lv = relicLv(k), ad = relicAd(k);
  if(ad >= RELIC_BREAK_MAX){ toast('已突破至化境（' + RELIC_BREAK_MAX + '/' + RELIC_BREAK_MAX + '）'); return; }
  if(lv < RELIC_LV_MAX){ toast('需先升满 ' + RELIC_LV_MAX + ' 级（当前 ' + lv + ' 级）'); return; }
  const fg = relicBreakFrag(ad, k), gm = relicBreakGem(ad, k);
  if(!canPay('relicFrag', fg)){ toast('碎片不足（需 ' + dStr(fg) + '，当前 ' + dStr(resD('relicFrag')) + '）'); return; }
  if(!canPay('gem', gm)){ toast('宝石不足（需 ' + dStr(gm) + '，当前 ' + dStr(resD('gem')) + '）'); return; }
  /* 必须检查 payCost 返回值：费用超出 SAFE_INT 时它返回 false 而不扣费。
     原实现忽略返回值直接推进突破次数 —— 一旦 canPay 与 payCost 的判定边界分叉
     （前者用大数比较、后者用 Number 比较），就会出现「没扣费却成功突破」的白嫖。 */
  if(!payCost('relicFrag', fg)){ toast('碎片扣除失败（费用超出可处理上限）'); return; }
  if(!payCost('gem', gm)){ resAdd('relicFrag', dNum(fg)); toast('宝石扣除失败，已退还碎片'); return; }
  S.relicAd = S.relicAd || {}; S.relicLv = S.relicLv || {};
  S.relicAd[k] = ad + 1;
  S.relicLv[k] = 0;                                    /* 突破后等级归零，重新累计 */
  const r = RELICS.find(function(x){ return x.k === k; });
  toast('【' + r.n + '】第 ' + S.relicAd[k] + ' 次突破！效果提升，等级重置');
  logMsg('【' + r.n + '】第 ' + S.relicAd[k] + ' 次突破（效果 ×' + nStr(1 + relicAdCoef(r.c) * S.relicAd[k]) + '），碎片 -' + nStr(fg) + '，宝石 -' + nStr(gm));
  renderRelic(); render(); save();
}
function craftRelic(k){

  toast('合成【' + r.n + '】！' + r.d);
  logMsg('合成遗物【' + r.n + '】：' + r.d);
  renderRelic(); render(); save();
}

/* ==================== 灵界法宝 ====================
   部位主属性刻意避开 攻速 / 暴击率 / 连击（三者有设计上限，不做随机放大）：
     法器→攻击　护甲→血量　玉佩→防御　戒指→暴伤　靴履→闪避　灵印→穿透

/* 1.80：6 件 → 9 件（正好 3×3 九宫格）。新增 头冠(命中) / 项链(灵力) / 腰带(灵石)。

/* 2.16：品阶由 5 档扩为 7 档（新增 圣品 / 道品）——

const ARTI_Q    = ['凡品','灵品','玄品','神品','仙品','圣品','道品'];
const ARTI_QC   = ['#9a9a9a','#4caf50','#2196f3','#9c27b0','#ffd54a','#ff8c2b','#ff5ea8'];
const ARTI_MAIN = [8, 18, 40, 85, 170, 340, 700];   /* 主属性基准%（随机 ±20%） */
const ARTI_AFFN = [1, 2, 3, 3, 4, 5, 6];            /* 词条条数 */
const ARTI_AFF_POOL = ['atk','hp','def','cdm','pen','dodge','hit','gp','gs','gu'];  /* 无 spd/crit/combo */

const ARTI_ENH_STEP = 0.08;                    /* 每级主属性 +8% */
const ARTI_ENH_ALL  = 0.35;                    /* 2.12：强化对「非主属性」的折算比例（全属性随强化一起涨） */
const ARTI_PCT_UNIT = {gp:'%', gs:'%', gu:'%', cdm:'%', atk:'%', hp:'%', def:'%', pen:'%', dodge:'', hit:''};
/* 品阶可达层数门槛：层数越高，妖王掉落的品阶上限越高 */
const ARTI_Q_GATE = [0, 50, 150, 400, 1000, 2500, 6000];
const ARTI_QW     = [31, 41, 18, 1.0, 0.22, 0.045, 0.009];    /* 2.18：低阶再略提、中高品质再次下调（神 −2.2 / 仙 −3 / 圣 −3.7 / 道 −4.7 倍） */
const ARTI_QW_FALL = 0.45;                     /* 档位衰减：层数越深，低阶权重衰减越多（否则第 1200 关仍大概率掉凡品） */
/* 合成：消耗碎片（符文）/ 洗练石（法宝）直接产出指定品阶，产物随机、强化与等级不继承。
   掉落是运气，合成是进度 —— 2.17 起高品质掉落大幅下调，合成成为主要获取途径。
   两套成本分开标定：各自按「分解 N 件同品阶」定价（法宝 ∝ 洗练石 100×1.8^q、符文 ∝ 碎片 2^q），
   2.18：中高品质合成成本大幅提高（折合分解同品阶：法宝 2/3/5/10/30/100/300 件、
   符文 20/30/45/80/180/500/1500 件），与「高品质主要靠合成获取」配套。 */
const FUSE_SHARD_ARTI = [200, 540, 1620, 5830, 31500, 189000, 1020000];     /* 洗练石：约 2 / 3 / 5 / 10 / 30 / 100 / 300 件同品阶分解 */
const FUSE_SHARD_RUNE = [20, 60, 180, 640, 2880, 16000, 96000];            /* 符文碎片：约 20 / 30 / 45 / 80 / 180 / 500 / 1500 件同品阶分解 */
/* 1.46 高阶惩罚：品阶每高一档，权重再乘 ARTI_QW_HIGHPEN。
   ★ 为什么需要：原公式只有「低阶衰减」FALL^(maxQ-i)，当 maxQ=4（仙品可达）时，
     低阶被压到与高阶同量级 —— 实测仙品 10.4% / 神品 18.8%，高阶太容易出。
     加入高阶惩罚后（0.5）：仙品 1.4% / 神品 5.1% / 玄品 11.5% / 灵品 26% / 凡品 56%。 */
const ARTI_QW_HIGHPEN = 0.25;   /* 2.17：0.35 → 0.25，高品质再降一档（神 −5.8 倍 / 仙 −6.8 倍 / 圣 −13.7 倍 / 道 −23 倍） */
function artiData(){
  if(!S.lmgArti) S.lmgArti = {equip:{}, bag:[], stone:0, seq:0};
  const d = S.lmgArti;
  if(!d.equip) d.equip = {};
  if(!d.bag) d.bag = [];
  d.stone = Math.max(0, d.stone || 0);
  d.seq = d.seq || 0;
  return d;
}
function artiSlotDef(k){ for(let i=0;i<ARTI_SLOTS.length;i++) if(ARTI_SLOTS[i].k===k) return ARTI_SLOTS[i]; return null; }
function artiMaxQ(lv){
  let q = 0;
  for(let i = 0; i < ARTI_Q_GATE.length; i++) if((lv||1) >= ARTI_Q_GATE[i]) q = i;
  return q;
}
function artiRndQ(maxQ){
  let tot = 0;
  const w = [];
  for(let i = 0; i <= maxQ; i++){
    /* 距当前可达上限越远的低阶，权重按 ARTI_QW_FALL^档位差 衰减；
       再叠加「高阶惩罚」HIGHPEN^品阶，使高品质掉落率显著低于原设计。 */
    w[i] = ARTI_QW[i] * Math.pow(ARTI_QW_FALL, maxQ - i) * Math.pow(ARTI_QW_HIGHPEN, i);
    tot += w[i];
  }
  let r = Math.random() * tot;
  for(let i = 0; i <= maxQ; i++){ r -= w[i]; if(r <= 0) return i; }
  return 0;
}
function artiRollAffix(q, main){
  const n = ARTI_AFFN[q] || 1, aff = [], used = {};
  for(let i = 0; i < n; i++){
    let k = ARTI_AFF_POOL[Math.floor(Math.random() * ARTI_AFF_POOL.length)];
    let guard = 0;
    while(used[k] && guard < 20){ k = ARTI_AFF_POOL[Math.floor(Math.random() * ARTI_AFF_POOL.length)]; guard++; }
    used[k] = 1;
    const v = Math.round(main * (0.30 + Math.random() * 0.30) * 10) / 10;   /* 主属性的 30%~60% */
    aff.push({k:k, v:Math.max(0.5, v)});
  }
  return aff;
}
/* 掉一件法宝：层数决定品阶上限 */
function artiMake(lv){
  return artiMakeByQ(artiMaxQ(lv || 1));
}
/* 按「品阶上限」造一件（心魔按成功次数决定品阶，与主线层数无关） */
function artiMakeByQ(maxQ){
  const d = artiData();
  const slot = ARTI_SLOTS[Math.floor(Math.random() * ARTI_SLOTS.length)];
  const q = artiRndQ(Math.max(0, Math.min(ARTI_Q.length - 1, maxQ)));
  const minQ = pickAutoQ('arti');
  if(minQ >= 0 && q < minQ){                    /* 低于设定品阶：就地分解成强化石 */
    const g = artiMeltVal(q, 0);
    d.stone = Math.min(SAFE_INT, (d.stone || 0) + g.shards);
    S.lmgStone = Math.min((S.lmgStone || 0) + g.stone, SAFE_INT);
    toast('低于设定品阶，已自动分解：强化石 +' + g.shards);
    return null;
  }
  if(d.bag.length >= PICK_MAX){
    toast('法宝背包已满（' + PICK_MAX + ' 件），无法拾取');
    return null;
  }
  const main = Math.round(ARTI_MAIN[q] * (0.8 + Math.random() * 0.4) * 10) / 10;
  const it = {id:++d.seq, slot:slot.k, q:q, main:main, enh:0, aff:artiRollAffix(q, main)};
  d.bag.push(it);
  return it;
}
/* 分解收益（供自动分解与手动分解共用，避免两处各写一份） */
function artiMeltVal(q, enh){
  return {
    stone:  Math.floor(100 * Math.pow(1.8, q) * (1 + (enh || 0) * 0.5)),
    shards: Math.max(1, q + Math.floor((enh || 0) / 5))
  };
}
/* ══════ 法宝合成（消耗洗练石）═════
   直接用洗练石产出指定品阶，产物随机生成、强化归零。
   刻意不继承强化：否则先强化再合等于按比例返还强化成本，强化系统的消耗意义会被削弱。 */
function artiMakeFixedQ(q){
  const d = artiData();
  const slot = ARTI_SLOTS[Math.floor(Math.random() * ARTI_SLOTS.length)];
  const main = Math.round(ARTI_MAIN[q] * (0.8 + Math.random() * 0.4) * 10) / 10;
  const it = {id:++d.seq, slot:slot.k, q:q, main:main, enh:0, aff:artiRollAffix(q, main)};
  d.bag.push(it);
  return it;
}
/* 当前可用的洗练石（合成与洗练共用同一资源） */
function artiShard(){ return Math.max(0, artiData().stone || 0); }
function artiFuseCost(q){ return FUSE_SHARD_ARTI[Math.max(0, Math.min(FUSE_SHARD_ARTI.length - 1, q || 0))]; }
function artiFuse(q, n){
  const d = artiData();
  n = Math.max(1, n || 1);
  const cost = artiFuseCost(q) * n;
  if(artiShard() < cost){ toast('洗练石不足（需 ' + nStr(cost) + '）'); return false; }
  if(d.bag.length + n > PICK_MAX){ toast('背包空间不足（' + PICK_MAX + ' 件）'); return false; }
  d.stone = Math.max(0, (d.stone || 0) - cost);
  let nm = '';
  for(let i = 0; i < n; i++){ const it = artiMakeFixedQ(q); if(!nm) nm = artiName(it); }
  toast('合成 ' + ARTI_Q[q] + '·' + nm + (n > 1 ? (' ×' + n) : ''));
  logMsg('洗练石合成法宝：' + ARTI_Q[q] + ' ×' + n + '（耗 ' + nStr(cost) + '）');
  renderArti(); renderNow(true); save();
  return true;
}

/* 单件对某属性的加成：主属性 ×(1+强化) + 词条 */
function artiItemVal(it, key){
  if(!it) return 0;
  let v = 0;
  const sd = artiSlotDef(it.slot);
  const enh = (it.enh || 0) * ARTI_ENH_STEP;
  let isMain = false, hasAff = false;
  if(sd && sd.key === key){ v += it.main * (1 + enh); isMain = true; }
  (it.aff || []).forEach(function(a){
    if(a.k === key){ v += a.v * (1 + enh * 0.5); hasAff = true; }   /* 强化对词条只给半额 */
  });
  /* 2.12：强化对「所有属性」生效 —— 非本槽主属性、又非词条的其余属性，
     按主属性基准 × ARTI_ENH_ALL 折算，强化越高涨得越多（不再只有主属性动）。 */
  if(!isMain && !hasAff && enh > 0) v += it.main * ARTI_ENH_ALL * enh;
  return v;
}
function artiPct(key){
  if(!key) return 0;
  const d = artiData();
  let v = 0;
  ARTI_SLOTS.forEach(function(sd){
    const it = artiEquipped(sd.k);
    if(it) v += artiItemVal(it, key);
  });
  return v * ARTI_PCT_NERF;   /* 1.68：法宝专属削弱（可单独调，取代全局系数） */
}
/* 1.81：强化 / 洗练消耗「按品阶拉开」。
   原强化系数 (1+q×0.8) 在仙品只有 ×4.2，而仙品主属性基准是凡品的 21 倍（ARTI_MAIN）——
   高品质同样便宜，等于把最强档白送。现改为 ×1/2.5/6/15/40，与品阶强度梯度对齐。 */
const ARTI_Q_COSTMUL = [1, 2.5, 6, 15, 40, 110, 300];
function artiEnhCost(it){
  const m = ARTI_Q_COSTMUL[Math.max(0, Math.min(ARTI_Q_COSTMUL.length - 1, it.q || 0))];
  return Math.floor(artiEnhCostAt(it.enh || 0) * m);
}
/* 1.84：上限放开到 +1000，原 1.35^lv 在 lv=1000 早已突破 SAFE_INT（1.35^1000 ≈ 1e130），
   故改为分段曲线：+0~15 沿用原手感（1.35^lv），+16~100 放缓（1.12^），
   +101 起改幂函数 (lv/100)^3 —— 既不溢出，又保持「每级更贵」的体感。
   实测（凡品 m=1）：+15 ≈ 1.8万、+100 ≈ 2.7亿、+500 ≈ 3.4e12、+1000 ≈ 2.7e13。 */
function artiEnhCostAt(lv){
  lv = Math.max(0, Math.floor(lv || 0));
  let c = 200;
  if(lv <= 15) c = 200 * Math.pow(1.35, lv);
  else{
    const c15 = 200 * Math.pow(1.35, 15);
    if(lv <= 100) c = c15 * Math.pow(1.12, lv - 15);
    else c = c15 * Math.pow(1.12, 85) * Math.pow(lv / 100, 3);
  }
  return Math.min(SAFE_INT, c);
}
const ARTI_REROLL_COST = [1, 2, 4, 8, 16, 32, 64];      /* 洗练石：原 1+floor(q/2) 在凡/灵/玄/神/仙 只差到 3 枚，梯度几乎不存在 */
function artiRerollCost(it){ return ARTI_REROLL_COST[Math.max(0, Math.min(ARTI_REROLL_COST.length - 1, it.q || 0))]; }
function artiName(it){

/* 1.81：法宝槽操作面板（与符文槽 runeSlotSheet 同一交互范式）
   点击已装备槽 → 列出「卸下 / 强化 / 洗练 / 分解」，不再一次点击就摘装备。 */
function artiSlotSheet(slot){
  const d = artiData(), sd = artiSlotDef(slot);
  if(!sd) return;
  const eqId = d.equip ? d.equip[slot] : undefined;
  const it = (eqId !== undefined && eqId !== null) ? artiById(eqId) : null;
  if(!it){
    /* 空槽：列出可装备的同部位法宝 */
    const free = (d.bag || []).filter(function(x){ return x.slot === slot && d.equip[slot] !== x.id; })
                              .sort(function(a, b){ return (b.q - a.q) || (b.main - a.main); });
    let h = '<div class="ish-nm">' + sd.n.replace(/\s/g,'') + '</div>'
          + '<div class="ish-dim">空置，可装备 1 件' + sd.n.replace(/\s/g,'') + '</div>';
    if(!free.length) h += '<div class="ish-line" style="margin-top:6px">背包中没有该部位的法宝。</div>';
    else{
      h += '<div class="ish-grp">可装备 ' + free.length + ' 件</div>';
      free.slice(0, 14).forEach(function(x){
        h += '<div class="ish-line" style="cursor:pointer" onclick="artiEquip(' + x.id + ');closeItemSheet()">'
          + '<b style="color:' + ARTI_QC[x.q] + '">' + ARTI_Q[x.q] + '·' + artiName(x) + '</b>　主属性 +'
          + (Math.round(artiItemVal(x, sd.key) * 10) / 10) + (ARTI_PCT_UNIT[sd.key] || '') + '</div>';
      });
    }
    openItemSheet('法 宝', h); return;
  }
  const key = sd.key;
  let aff = '';
  const _affK = {};
  (it.aff || []).forEach(function(a){
    _affK[a.k] = 1;
    aff += '<div class="ish-line">' + (ARTI_AFF_NAME[a.k] || a.k) + ' +' + a.v + (ARTI_PCT_UNIT[a.k] || '') + '</div>';
  });
  /* 2.16：强化对全部属性生效 —— 其余属性直接接在词条后面逐条列出（与主属性同款行），
     不再在底部单独写一句「其余属性各 +X」的说明（玩家看不到具体是哪些属性）。 */
  ARTI_AFF_POOL.forEach(function(k2){
    if(k2 === key || _affK[k2]) return;
    const _v2 = artiItemVal(it, k2);
    if(!(_v2 > 0)) return;
    aff += '<div class="ish-line" style="color:#8a8070">' + (ARTI_AFF_NAME[k2] || k2) + ' +'
         + (Math.round(_v2 * 10) / 10) + (ARTI_PCT_UNIT[k2] || '') + '</div>';
  });
  const h = '<div class="ish-nm" style="color:' + ARTI_QC[it.q] + '">' + (it.enh > 0 ? ('+' + it.enh + ' ') : '') + artiName(it) + '</div>'
    + '<div class="ish-dim">' + sd.n.replace(/\s/g,'') + ' · ' + ARTI_Q[it.q] + ' · 强化 +' + (it.enh || 0) + ' / ' + ARTI_ENH_MAX + '</div>'
    + '<div class="ish-line">主属性：' + (ARTI_AFF_NAME[key] || key) + ' +'
      + (Math.round(artiItemVal(it, key) * 10) / 10) + (ARTI_PCT_UNIT[key] || '') + '</div>'
    + aff
    + '<div class="ish-ops">'
    + '<div class="fs-btn b-brown" onclick="artiDoUnequip(\'' + slot + '\')"><span>卸 下</span></div>'
    + '<div class="fs-btn b-gold" onclick="artiDoEnh(' + it.id + ')"><span>强化 ' + nStr(artiEnhCost(it)) + '</span></div>'
    + '<div class="fs-btn b-blue" onclick="artiDoReroll(' + it.id + ')"><span>洗练 ' + nStr(artiRerollCost(it)) + '</span></div>'
    + '<div class="fs-btn b-red" onclick="artiDoMelt(' + it.id + ')"><span>分 解</span></div>'
    + '</div>';
  openItemSheet('法 宝', h);
}
function artiDoUnequip(slot){ artiUnequip(slot); closeItemSheet(); }

function artiDoReroll(id){ artiReroll(id); artiSlotSheet(artiById(id) ? artiById(id).slot : ''); }
function artiDoMelt(id){ const s0 = artiById(id) ? artiById(id).slot : ''; artiMelt(id); if(s0) artiSlotSheet(s0); else closeItemSheet(); }
function artiUnequip(slot){
  const d = artiData();
  if(!d.equip[slot]) return;
  const it = artiById(d.equip[slot]);
  delete d.equip[slot];
  if(it) toast('已卸下 ' + artiName(it));
  renderArti(); renderNow(true); save();
}
function artiEnh(id){
  const it = artiById(id);
  if(!it) return;
  if((it.enh || 0) >= ARTI_ENH_MAX){ toast('已达强化上限 +' + ARTI_ENH_MAX); return; }
  const c = artiEnhCost(it);
  if((S.lmgStone || 0) < c){ toast('灵石不足（需 ' + nStr(c) + '）'); return; }
  S.lmgStone -= c; it.enh = (it.enh || 0) + 1;
  toast(artiName(it) + ' 强化至 +' + it.enh);
  logMsg('法宝强化：' + artiName(it) + ' → +' + it.enh + '（消耗灵石 ' + nStr(c) + '）');
  renderArti(); renderNow(true); save();
}
function artiReroll(id){
  const d = artiData(), it = artiById(id);
  if(!it) return;
  const c = artiRerollCost(it);
  if((d.stone || 0) < c){ toast('洗练石不足（需 ' + c + '）'); return; }
  d.stone -= c;
  it.aff = artiRollAffix(it.q, it.main);
  toast(artiName(it) + ' 洗练完成');
  logMsg('法宝洗练：' + artiName(it) + '（消耗洗练石 ' + c + '）');
  renderArti(); renderNow(true); save();
}
function artiMelt(id){
  const d = artiData(), it = artiById(id);
  if(!it) return;
  if(d.equip[it.slot] === it.id) delete d.equip[it.slot];
  const _v = artiMeltVal(it.q, it.enh || 0);
  const stone = _v.stone, shards = _v.shards;
  S.lmgStone = Math.min((S.lmgStone || 0) + stone, SAFE_INT);
  d.stone = Math.min(SAFE_INT, (d.stone || 0) + shards);
  for(let i = 0; i < d.bag.length; i++) if(d.bag[i].id === it.id){ d.bag.splice(i, 1); break; }
  toast('分解 ' + artiName(it) + '：灵石 +' + nStr(stone) + '　洗练石 +' + shards);
  logMsg('分解法宝【' + artiName(it) + '】：灵石 +' + nStr(stone) + '　洗练石 +' + shards);
  renderArti(); renderNow(true); save();
}

/* ==================== 心魔试炼 ====================
   与主线「灵界妖王」彻底分开的独立挑战：
   · 名称：心魔（主线每 10 关那只仍叫「灵界妖王」）

   · 进入时保存主线层数，退出 / 战败均恢复；心魔战不推进也不损失主线进度
   · 次数无上限：能打过就一直打，每赢一次心魔更强、掉落品阶更高 */
/* 1.43 心魔加强：底数与成长率同步提高，等效层数步长翻倍。
   配合「法宝 / 符文掉落大幅降低」，心魔成为两者的主要稳定来源，
   故它必须更难 —— 否则玩家刷心魔就能轻松拿满，掉落削减失去意义。 */
/* 1.66 心魔加强：底数 ×2.25、成长 1.45→1.52 / 1.40→1.48、等效层数步长 4→7。
   原参数下心魔偏软（胜 10 次才 33 万血、2600 攻），玩家几乎无阻力地一路刷满，
   法宝 / 符文的掉落削减因此失去意义。现胜 10 次约 130 万血、1.3 万攻。 */
const TRIAL_BASE_HP  = 18000, TRIAL_HP_GROW  = 1.52;  /* 心魔血量：底数 × 成长^成功次数 */

/* 心魔被击败：成功次数 +1 → 下一只更强 */
function lmgTrialClear(){
  S.lmgTrialWin = lmgTrialWin() + 1;
  const d = artiData();
  /* 1.56：法宝由必掉改为概率掉落（LMG_TRIAL_ARTI_RATE），洗练石仍必掉。 */
  /* 2.16：改为「全品阶随机」—— 原按成功次数封顶，心魔刷到后期品阶必然触顶，
     掉落变成确定性产物；随机全品阶后，高阶永远是小概率惊喜。 */
  const it = (Math.random() < LMG_TRIAL_ARTI_RATE) ? artiMakeByQ(ARTI_Q.length - 1) : null;

     1.65：由「每次必掉」改为概率掉落，与法宝同口径 —— 心魔不再是符文印钞机。 */
  const _rt = (Math.random() < LMG_TRIAL_RUNE_RATE) ? runeMakeByQ(RUNE_Q.length - 1) : null;
  /* 1.48：成就「心魔通关」→ 心魔奖励 +15%/阶段（线性作用于灵力与灵石） */
  const _xmMul = 1 + achXmPct() / 100;
  S.lmgPower = dadd(_toD(S.lmgPower || D(0)), dmul(D(200 * _xmMul), dpow(D(1.18), lmgTrialWin())));
  S.lmgStone = Math.min((S.lmgStone || 0) + Math.floor(500 * _xmMul * Math.pow(1.16, lmgTrialWin())), SAFE_INT);
  const _artiTxt = it ? ('　' + artiName(it)) : '';
  toast('心魔伏诛！第 ' + lmgTrialWin() + ' 次成功' + _artiTxt + '　洗练石 ×' + sh);
  logMsg('心魔试炼第 ' + lmgTrialWin() + ' 次成功：获得' + (it ? '【' + artiName(it) + '】' : '洗练石') + ' ×' + sh + '（下一只更强）');
  if(_rt) logMsg('心魔掉落符文：【' + RUNE_Q[_rt.q] + '·' + runeKindDef(_rt.k).n + '】');
  /* 立即刷新下一只心魔（不退出、不推进主线） */
  S.lmgMonMaxHp = lmgTrialHp();
  S.lmgMonHp = S.lmgMonMaxHp;
  const mx = lmgAttrD('hp');
  S.lmgHp = dmin(mx, dadd(_toD(S.lmgHp || D(0)), dmul(mx, D(0.35))));
  renderNow(true); save();
}

/* ==================== 灵界训练洞府 ====================
   模型：多产线独立投资。洞府自带货币「洞府灵玉」，由洞府自己产出，
   升级只花灵玉 —— 与灵石 / 灵力分隔，形成「洞府赚钱 → 投回洞府 → 赚更多」的闭环，
   不会因为洞府而挤占灵界主线养成。
   成本指数 ×1.15/级、产出线性 +基准/级：存在净收益最优停点，不是无脑点到底，
   但也刻意不做硬上限，想无限投就无限投（等级无上限）。 */
const CAVE_LINES = [

     原灵玉只由「总等级 ×0.25」被动产出，开局 0.6/时 极慢，第一条产线都要攒很久。
     此产线让灵玉成为一条可主动投资的增长线，形成「投灵玉 → 产更多灵玉」的加速循环。
     成本仍是 10 × 1.15^等级（指数）而产出线性，故存在最优停点，不会失控。 */
  /* 1.64：各产线 base 累计上调 50 倍（1.56 只上调 50%，洞府依旧「点满就忘」） */
  {k:'jade',  n:'灵　玉', ic:'◍', base:112.5, unit:'/时'},
  {k:'power', n:'灵力',   ic:'✦', base:225,   unit:'/时'},
  {k:'stone', n:'灵石',   ic:'◈', base:150,   unit:'/时'},
  {k:'soul',  n:'灵魂',   ic:'✧', base:112.5, unit:'/时'},
  {k:'pexp',  n:'灵兽经验', ic:'❋', base:3000, unit:'/时'},
  {k:'plexp', n:'主角经验', ic:'✵', base:2250, unit:'/时'},
  /* —— 凡人产线：不直接产资源，而是每小时掷骰掉「凡人界物品」——
     命中后用凡尘宝箱同一套池（越珍贵概率越低），已拥有则走重复突破。
     概率刻意远低于开宝箱（每级仅 +0.004%/时），属于长线添头，不冲击宝箱经济。 */
  {k:'neili', n:'内　力',   ic:'❂', base:600,  unit:'/时'},
  {k:'gem',   n:'宝　石',   ic:'◈', base:400,  unit:'/时'},
  {k:'pts',   n:'转生点',   ic:'✥', base:12,   unit:'/时'},
  /* —— 灵脉产线：直接产「灵脉」系统的养成资源（矿石 / 灵蕴），用洞府灵玉升级。
     灵脉本身靠占领矿脉产出，此产线是额外的稳定补充，不与矿脉争夺 ——
     即使一处矿都没占到，灵脉节点也有缓慢的成长来源。 */
  {k:'ore',   n:'凡脉·矿石', ic:'◆', base:60,  unit:'/时'},
  {k:'ly',    n:'灵脉·灵蕴', ic:'◇', base:40,  unit:'/时'}
];
const CAVE_RARE_KIND = {tre:'tre', pet:'pet', gf:'gf'};
/* 凡人产线掷骰：返回命中的物品（未命中返回 null） */
function caveRareRoll(kind, pPerHour, hrs){
  if(!(pPerHour > 0) || !(hrs > 0)) return null;
  const p = Math.min(0.9, pPerHour / 100 * hrs);      /* base 是「%/时」 */
  if(Math.random() >= p) return null;
  const pool = boxRarePool().filter(function(x){ return x.K.kind === kind; });
  if(!pool.length) return null;
  let tot = 0; pool.forEach(function(x){ tot += x.p; });
  if(!(tot > 0)) return null;
  let r = Math.random() * tot;
  for(let i = 0; i < pool.length; i++){ r -= pool[i].p; if(r <= 0) return pool[i]; }
  return pool[pool.length - 1];
}
const CAVE_COST_BASE = 8, CAVE_COST_GROW = 1.12;   /* 1.64：10×1.15^lv → 8×1.12^lv，升级更便宜、能推更高等级 */
function caveData(){
  if(!S.lmgCave) S.lmgCave = {lv:{}, jade:0, acc:0};
  const d = S.lmgCave;
  if(!d.lv) d.lv = {};
  d.jade = Math.max(0, d.jade || 0);
  return d;
}
function caveLv(k){ const d = caveData(); return Math.max(0, d.lv[k] || 0); }
function caveLineDef(k){ for(let i=0;i<CAVE_LINES.length;i++) if(CAVE_LINES[i].k===k) return CAVE_LINES[i]; return null; }
/* 某产线每小时产出（未含全局加成） */
/* 1.56 洞府加强：
   原产出 = base × 等级，纯线性 —— 前期尚可，中后期相对灵界资源消耗量几乎可忽略，
   洞府沦为「点满就忘」的摆设。改为 base × 等级 × 等级成长系数，
   并上调各产线 base，让洞府重新成为一条值得持续投资的线。
   CAVE_LV_BOOST 越小成长越快：默认 30，即每满 30 级该产线产出翻倍。 */
const CAVE_LV_BOOST = 15;   /* 1.64：30 → 15，每满 15 级产出翻倍（成长更快） */
function caveLinePH(k){
  const d = caveLineDef(k);
  if(!d) return 0;
  const lv = caveLv(k);
  return d.base * lv * (1 + lv / CAVE_LV_BOOST);
}
/* 灵玉产出：只与「洞府总等级」挂钩，与各产线分配无关 —— 投入越多，赚钱越快 */
function caveTotalLv(){ let t = 0; CAVE_LINES.forEach(function(l){ t += caveLv(l.k); }); return t; }
/* 灵玉产出 = 保底（0.6 + 洞府总等级 ×0.25）+ 灵玉产线产出。
   注：caveTotalLv() 已含灵玉产线自身等级，故该产线每级实际贡献 = base(1.0) + 0.25。 */
function caveJadePH(){ return 30 + caveTotalLv() * 8 + caveLinePH('jade'); }   /* 1.64：保底 1.2→30，总等级系数 0.5→8 */
function caveUpCost(k){
  const lv = caveLv(k);
  if(lv === 0) return 0;                            /* 第一条产线免费：给洞府一个零门槛起点 */
  return Math.floor(CAVE_COST_BASE * Math.pow(CAVE_COST_GROW, lv - 1));   /* 首级免费后，成本从第 2 级起算 */
}
function caveUp(k){
  const d = caveData(), c = caveUpCost(k);
  if((d.jade || 0) < c){ toast('洞府灵玉不足（需 ' + nStr(c) + '）'); return false; }
  d.jade -= c;
  d.lv[k] = caveLv(k) + 1;
  toast((caveLineDef(k) || {}).n + '产线 → Lv.' + d.lv[k]);
  logMsg('洞府升级：【' + (caveLineDef(k) || {}).n + '】产线 Lv.' + d.lv[k] + '（消耗洞府灵玉 ' + nStr(c) + '）');
  renderCave(); save();
  return true;
}
/* 洞府总览：每小时各项产出 */
function caveGainPH(){
  const g = {};
  CAVE_LINES.forEach(function(l){ g[l.k] = caveLinePH(l.k); });
  g.jade = caveJadePH();
  return g;
}
/* 结算 hrs 小时产出（在线 / 离线共用）。sec 用于小数累计，避免短间隔被抹成 0 */
function caveApply(hrs){
  if(!S.lmgUnlocked) return null;
  if(!(hrs > 0)) return null;
  const g = caveGainPH();
  /* 不能因为「所有产线都是 0 级」就直接 return null：
     洞府灵玉有 0.6/时 的保底产率（不依赖产线等级），若此时不结算，
     玩家永远攒不到第一笔灵玉、也就永远升不了第一条产线 —— 洞府开局即死锁。
     判据改为：灵玉与产线全为 0 才算无产出。 */
  if(!(g.jade > 0) && !CAVE_LINES.some(function(l){ return g[l.k] > 0; })) return null;   /* 凡人产线 base 即产率，同样 >0 */
  const d = caveData();
  const r = {jade:0, power:0, stone:0, soul:0, pexp:D(0), plexp:D(0), neili:0, gem:0, pts:D(0)};
  /* 灵玉：允许小数累积（d.acc），否则低级洞府每小时 0.6 玉会被 Math.floor 抹成 0 */
  const jadeRaw = (d.acc || 0) + g.jade * hrs;      /* 只累计零头 + 本次新增，d.jade 已是整数部分 */
  r.jade = Math.floor(jadeRaw);
  d.acc = jadeRaw - r.jade;
  d.jade = (d.jade || 0) + r.jade;
  r.power = Math.floor(g.power * hrs);
  if(r.power > 0) S.lmgPower = dadd(_toD(S.lmgPower || D(0)), D(r.power));
  r.stone = Math.floor(g.stone * hrs);
  if(r.stone > 0) S.lmgStone = Math.min((S.lmgStone || 0) + r.stone, SAFE_INT);
  r.soul = Math.floor(g.soul * hrs);
  if(r.soul > 0){
    /* 灵魂是普通 Number（上限 SAFE_INT），绝不能走 dadd/_toD —— 写进去就变成大数对象，
       之后所有「(S.soul||0) + n」退化成字符串拼接，再被 capRes 的 !isFinite 清成 0。
       统一走 capRes('soul', 普通数字加法)。 */
    resAdd('soul', soulGain(r.soul));
    r.soul = Math.floor(r.soul);
  }
  if(g.pexp > 0){
    r.pexp = dmul(D(g.pexp), D(hrs));
    S.lmgPetExp = dadd(petExpPool(), r.pexp);
  }
  if(g.plexp > 0){
    r.plexp = dmul(D(g.plexp), D(hrs));
    plvAddExp(r.plexp);
  }
  /* 灵脉产线：矿石 / 灵蕴 —— 进灵脉系统资源（与矿脉占领产出同源、同上限） */
  r.ore = Math.floor((g.ore || 0) * hrs);
  r.ly  = Math.floor((g.ly  || 0) * hrs);
  if(r.ore > 0 || r.ly > 0){
    const m = maiData();
    if(r.ore > 0) m.ore = Math.min(SAFE_INT, (m.ore || 0) + r.ore);
    if(r.ly  > 0) m.ly  = Math.min(SAFE_INT, (m.ly  || 0) + r.ly);
  }
  /* 内力 / 宝石 / 转生点 产线：直接进对应资源（内力、转生点为大数，宝石为普通数字） */
  r.neili = Math.floor((g.neili || 0) * hrs);
  if(r.neili > 0) nlAdd(r.neili);
  r.gem = Math.floor((g.gem || 0) * hrs);
  if(r.gem > 0) resAdd('gem', r.gem);
  if(g.pts > 0){
    r.pts = dmul(D(g.pts), D(hrs));
    /* S.pts 本身即大数对象，不能再用 D() 包一层（否则 dmul/dadd 拿到的是包装后的错误结构，
       表现为「结算有产出但转生点不涨」）。写法与凡人塔通关 S.pts = dadd(S.pts || D(0), gain) 一致。 */
    S.pts = dadd(S.pts || D(0), r.pts);
  }
  /* 原「凡人产线」（掷骰掉宝物 / 宠物 / 功法）已移除，改为上述三条资源产线 */
  CAVE_LINES.forEach(function(l){
    if(!l.rare) return;
    const hit = caveRareRoll(l.rare, g[l.k], hrs);
    if(!hit) return;
    if(hit.up){
      hit.lv = dupUp(hit.K.kind, hit.it.k);
      toast('洞府产出：' + hit.K.n + '【' + hit.it.n + '】突破至 ' + hit.lv + ' 重');
      logMsg('训练洞府产出：' + hit.K.n + '【' + hit.it.n + '】突破至 ' + hit.lv + ' 重');
    }else{
      hit.K.get(hit.it.k);
      toast('洞府产出：' + hit.K.n + '【' + hit.it.n + '】');
      logMsg('训练洞府产出：' + hit.K.n + '【' + hit.it.n + '】');
    }
    r.rare = (r.rare || 0) + 1;
  });
  return r;
}

/* ============ 凡尘宝箱 ============ */
const BOX_ALL_MAX = 2e5;            /* 「全部开启」单次上限：防止极端数量（百万级）卡死主线程 */
/* 凡尘宝箱常规掉落三档：丹药 / 遗物碎片 / 宝石，概率互斥且必须首尾相接（宝石为兜底 else）。
   2.18：丹药由 14% 提到 18%（枚数 1~2 → 1~3），碎片档同步右移以保持宝石 60% 不变。 */
const BOX_PILL_P = 0.18;
const BOX_FRAG_P = 0.22;
/* ============ 凡尘宝箱 · 稀有掉落（灵宠 / 功法 / 宝物） ============
   设计：每开 1 个宝箱独立掷骰，命中率极低；命中后再按「珍贵度权重」挑一件尚未拥有的。
   · 权重按售价分 6 档，越贵权重越低（每档约 ×0.18），最贵与最便宜相差数千倍
   · 已拥有的不再进池 → 不会重复掉落，且随着收集变全，剩下的都是珍贵的，出货越来越难
   · 1.81：不再有任何保底 —— 纯权重抽取，开多少箱都可能不出（越珍贵越难得） */
const BOX_RARE_P = 0.0005;                           /* 档位基准（各件概率 = 档位基准 × 类别系数，见 BOX_RARE_PT） */
/* 1.81：凡人宝箱稀有掉落保底 —— 已彻底移除（用户要求）。
   此前 1.59 加过「软保底（连开未出 800 箱后每 800 箱 +40%，上限 ×6）+ 硬保底（12000 箱必出）」。
   现全部删除：稀有掉落回到纯权重抽取，越珍贵越难得，不再有「开够数就必出」的兜底。
   S.boxPity 字段仅为兼容旧存档保留，不参与任何计算，也不再在面板显示。 */

/* ============ 重复掉落 → 叠加升级 ============
   · 已拥有的功法 / 灵宠 / 宝物仍会掉落，每重复掉 1 次「重数 +1」，效果按重数线性放大
   · 每重 +50% 基础效果（DUP_STEP），最多 10 重（DUP_MAX），满重后不再进池
   · 特殊属性（攻速 / 暴击率 / 连击）本身有上限，重复掉落不放大这些项 —— 只涨其余项
     例：疾风步（纯攻速）重复掉落不再进池；破煞真言（最终伤害）可升；
         洞玄天眸（暴击率 +40% / 暴伤 +300%）只涨暴伤，暴击率维持 40% */
const DUP_MAX  = 10;                                 /* 每件最多 10 重 */
const DUP_STEP = 0.5;                                /* 每重 +50% 基础效果 */
const DUP_SKIP = {spd:1, crit:1, combo:1};           /* 特殊属性：有上限，不参与升级 */
const DUP_KEY  = {gf:'gfLv', pet:'petLv', tre:'treLv'};
function dupLv(kind, k){ const m = S[DUP_KEY[kind]]; return (m && m[k]) || 0; }
function dupMul(kind, k, key){ if(DUP_SKIP[key]) return 1; return 1 + dupLv(kind, k) * DUP_STEP; }
function dupUp(kind, k){
  const key = DUP_KEY[kind];
  if(!S[key]) S[key] = {};
  const lv = Math.min(DUP_MAX, (S[key][k] || 0) + 1);
  S[key][k] = lv;
  return lv;
}
/* 该件重复掉落是否还能变强：未满重 且 至少有一项非特殊属性 */
function itemCanUp(kind, it){
  if(dupLv(kind, it.k) >= DUP_MAX) return false;
  if(kind === 'gf') return !DUP_SKIP[it.key];
  const e = (kind === 'pet') ? it.eff : it.e;
  for(const key in e) if(!DUP_SKIP[key]) return true;
  return false;
}
function dupTag(kind, k){                            /* 面板角标：仅已拥有且重数 > 0 时显示 */
  const lv = dupLv(kind, k);
  return lv > 0 ? '<span style="color:#ffd54a;font-size:11px"> ●' + lv + '重</span>' : '';
}
/* 6 档「单件」掉落概率：<1万 / <10万 / <100万 / <1000万 / <1亿 / ≥1亿
   2.18：整体下调，且高价（高品）档削得更狠 —— 低档约 ×0.75，顶档 ×0.06，稀有物主要靠合成 / 其它途径获取 */
const BOX_RARE_PT   = [1.5e-5, 9e-6, 1.4e-6, 1.6e-7, 2.2e-8, 2.5e-9];
function boxRareTier(cost){
  const c = cost || 0;
  if(c < 1e4) return 0;  if(c < 1e5) return 1;  if(c < 1e6) return 2;
  if(c < 1e7) return 3;  if(c < 1e8) return 4;  return 5;
}
const BOX_RARE_KIND = [
  {kind:'pet', n:'灵宠', w:0.5, list:function(){ return (typeof PETS      !== 'undefined') ? PETS      : []; },
   own:function(k){ return petBought(k); },
   get:function(k){ if(!S.pets) S.pets = {}; S.pets[k] = true; if(S.petsOff) delete S.petsOff[k]; }},
  {kind:'gf',  n:'功法', w:1.0, list:function(){ return (typeof GONGFA    !== 'undefined') ? GONGFA    : []; },
   own:function(k){ return gfBought(k); },
   get:function(k){ if(!S.gongfa) S.gongfa = {}; S.gongfa[k] = true; if(S.gongfaOff) delete S.gongfaOff[k]; }},
  {kind:'tre', n:'宝物', w:0.8, list:function(){ return (typeof TREASURES !== 'undefined') ? TREASURES : []; },
   own:function(k){ return treHas(k); },
   get:function(k){ S.treasures = S.treasures || {}; S.treasures[k] = true; }}
];
function boxRarePool(){
  const pool = [];
  for(let i = 0; i < BOX_RARE_KIND.length; i++){
    const K = BOX_RARE_KIND[i], list = K.list();
    for(let j = 0; j < list.length; j++){
      const it = list[j];
      if(K.own(it.k) && !itemCanUp(K.kind, it)) continue;   /* 已拥有且不能再升级：不再进池 */
      pool.push({K:K, it:it, up:K.own(it.k), p:BOX_RARE_PT[boxRareTier(it.cost)] * K.w});
    }
  }
  pool.sort(function(a, b){ return b.p - a.p; });     /* 概率降序：展示时「最易出 / 最难出」直接取首尾 */
  return pool;
}
function boxRareTotal(pool){ let t = 0; for(let i = 0; i < pool.length; i++) t += pool[i].p; return t; }   /* 本箱命中任意一件的概率 = 各件之和 */
/* 概率文案：极小概率写成「1 / N」，否则写百分比 */
function boxRareOdds(p){
  if(!(p > 0)) return '—';
  const inv = 1 / p;
  return inv >= 10000 ? ('1 / ' + nStr(Math.round(inv))) : ((Math.round(p * 1e6) / 1e4) + '%');
}

function openBox(all){
  if(dcmp(resD('box'), D(1)) < 0){ toast('没有凡尘宝箱'); return; }
  /* 「全部开启」名实相符：原实现硬编码最多只开 200 个，玩家有几万个宝箱时要点到手酸，
     且弹窗仍写「全部开启」属于功能与文案不符。
     现一次性开完；仅当数量极端（> 单次上限）时按上限分批，并在提示里说明剩余量。 */
  const cnt = all ? Math.min(resNum('box'), BOX_ALL_MAX) : 1;
  let gem = 0, frag = 0;
  const pillGot = {};
  /* 稀有掉落：整批共用一个池（避免每箱重建，万箱级开启也不卡），抽中即出池，同批不重复 */
  const pool = boxRarePool();
  let poolP = boxRareTotal(pool);                     /* 随池内剩余而变：越收集越难出 */
  const rareGot = [];
  const bm = boxMul();                                  /* 统一用 boxMul()（含成就/宠物/经脉/宝物），原仅算成就且末尾又乘一次 boxMul 导致叠加 */
  for(let i = 0; i < cnt; i++){
    const r = Math.random();
    if(r < BOX_PILL_P){                                                       /* 丹药 1~3 枚（随机种类），概率与枚数均可整体标定 */
      const pk = PILLS[Math.floor(Math.random() * PILLS.length)].k;
      const pn = 1 + (Math.random() < 0.50 ? 1 : 0) + (Math.random() < 0.20 ? 1 : 0);
      pillsAdd(pk, pn);                                                                          /* 灵丹改大数累加 */
      pillGot[pk] = (pillGot[pk] || 0) + pn;
    }else if(r < BOX_PILL_P + BOX_FRAG_P){ frag += Math.round((2 + (Math.random() < 0.30 ? 1 : 0)) * bm * (1 + relicVal('frag') / 100)); }  /* 遗物碎片 2~3（1.81：接入「拾遗遗物」） */
    else { gem += Math.round((6 + Math.floor(Math.random() * 13)) * bm); }                       /* 60% 宝石 6~18（原 47% / 2~8） */
    if(pool.length){                                    /* 稀有掉落：与上面的常规掉落相互独立 */
      /* 1.81：保底已移除 —— 命中率就是纯权重 poolP，不再随连开次数提升 */
      const hitP = Math.min(1, poolP);
      if(Math.random() < hitP){
        let r2 = Math.random() * poolP, pick = pool.length - 1;
        for(let j = 0; j < pool.length; j++){ r2 -= pool[j].p; if(r2 <= 0){ pick = j; break; } }
        const hit = pool[pick];
        poolP -= hit.p; pool.splice(pick, 1);           /* 出池：同批不会重复掉落同一件 */
        if(hit.up){                                     /* 重复掉落 → 重数 +1（特殊属性不涨） */
          hit.lv = dupUp(hit.K.kind, hit.it.k);
          if(hit.lv >= DUP_MAX){ hit.full = true; }     /* 满重：本次之后不再进池 */
        }else{
          hit.K.get(hit.it.k);                          /* 首次获得：立即生效（自动启用 / 出战） */
        }
        rareGot.push(hit);
        S.stRare = (S.stRare || 0) + 1;
      }
    }
  }
  /* bm 已在逐次结算中生效，此处不再重复乘 boxMul() */

  /* 2.23：box 已改为大数存储，减法由 resPay 的大数层完成，不再需要「先夹到安全整数」的老防御 */
  resPay('box', cnt);
  resAdd('gem', gem); resAdd('relicFrag', frag);
  S.stBox = (S.stBox || 0) + cnt;
  let pt = '';
  for(const k in pillGot){ pt += (pt ? '、' : '') + pillName(k) + ' ×' + pillGot[k]; }
  const rest = resD('box');                        /* 扣减后剩余 */
  const more = dcmp(rest, D(0)) > 0 ? ('，剩余 ' + dStr(rest) + ' 个可再点') : '';
  let rt = '';
  rareGot.forEach(function(h){
    rt += (rt ? '、' : '') + h.K.n + '【' + h.it.n + '】' + (h.up ? (' 突破至 ' + h.lv + ' 重') : '');
    logMsg('凡尘宝箱稀有掉落：' + h.K.n + '【' + h.it.n + '】' + (h.up ? (' 突破至 ' + h.lv + ' 重（效果 +' + Math.round(h.lv * DUP_STEP * 100) + '%，特殊属性不涨）') : (h.it.d ? '（' + h.it.d + '）' : '')));
  });
  toast('开启 ' + nStr(cnt) + ' 个宝箱：宝石 +' + nStr(gem) + '，遗物碎片 +' + nStr(frag) + (pt ? '，丹药 ' + pt : '') + (rt ? '　★稀有掉落：' + rt : '') + more);
  /* 1.56：「已自动启用」措辞过时（启用 / 停用已废除），改为「已永久生效」 */
  if(rt) toast('★ 稀有掉落：' + rt + '（已永久生效）');
  logMsg('开启凡尘宝箱 ×' + nStr(cnt) + '：宝石 +' + nStr(gem) + '，遗物碎片 +' + nStr(frag) + (pt ? '，丹药 ' + pt : '') + more);
  renderBag(); render();
  if(rareGot.length){ try{ renderPet(); renderGongfa(); renderTreasure(); }catch(e){} }
  save();
}

/* ============ 背包 ============ */
let bagCat = 'seed';                       /* 背包当前分类：seed 种子 / pill 丹药 / sup 辅助 / relic 遗物 */
function openBag(){ renderBag(); $('bagOverlay').classList.add('show'); }
function closeBag(){ $('bagOverlay').classList.remove('show'); }
function switchBagCat(c){
  bagCat = c;
  renderBagTabs();
  renderBag();
}
function renderBagTabs(){
  const t = $('bagTabs');
  if(!t) return;
  const defs = [
    {c:'seed',  n:'种子'},
    {c:'pill',  n:'丹药'},
    {c:'sup',   n:'辅助'}
  ];
  let h = '';
  defs.forEach(function(d){
    let cnt = '';
    if(d.c === 'seed'){ let n = 0; SEEDS.forEach(function(sd, i){ n += (S.seeds && S.seeds[i]) || 0; }); cnt = n ? ' (' + n + ')' : ''; }
    if(d.c === 'sup'){ const n = fertTotal(); cnt = n ? ' (' + n + ')' : ''; }
    h += '<div class="sh-tab' + (bagCat === d.c ? ' active' : '') + '" data-c="' + d.c + '" onclick="switchBagCat(\'' + d.c + '\')">' + d.n + cnt + '</div>';
  });
  t.innerHTML = h;
}

  $('bagGem').textContent = dStr(resD('gem'));

  let sh = '', st = 0;
  SEEDS.forEach(function(sd, i){ const c = (S.seeds && S.seeds[i]) || 0; st += c; if(c) sh += sd.n + ' ×' + c + '　'; });
  let ph = '', pt = D(0);
  PILLS.forEach(function(p){
    const c = pillsD(p.k);
    if(dcmp(c, D(0)) > 0){ pt = dadd(pt, c); ph += p.n + ' ×' + dStr(c) + '　'; }
  });
  let h = '';
  renderBagTabs();
  if(bagCat === 'seed'){

    h += '<div class="wp-tip">' + (sh || '暂无种子（秘境 / 商店）') + '</div>';

      const c = (S.seeds && S.seeds[i]) || 0;

           ' · 宝石 ' + nStr(farmGemOut(i)) + ' · 随机丹药 ×' + farmPillOut(i) + '</div></div>' +

  }else if(bagCat === 'pill'){

    PILLS.forEach(function(p){
      const hv = pillsD(p.k), has = dcmp(hv, D(0)) > 0;

  const rb = $('bagRareTip');
  if(rb){

    if(!p2.length){
      rb.innerHTML = '稀有掉落：<span style="color:#7ee0a0">灵宠 / 功法 / 宝物已全部集齐且满 ' + DUP_MAX + ' 重</span>（累计稀有掉落 ' + nStr(S.stRare || 0) + ' 件）';
