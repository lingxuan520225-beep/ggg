/* ══════════════════════════════════════════════════════════════
   香火：两界联动 —— 把修行成果凝为香火，反哺凡人界

   为什么需要它：FR_BOND 是「凡人界 6 项养成 → 灵界属性」，方向单一且算完即静止；
   灵界对凡人界则完全没有反馈，启灵之后凡人界等于退休。香火补的就是这条反向回路，
   与 FR_BOND 互为镜像。

   获取（无上限，可重复刷）：
     · 每次击杀 → 微量香火（随当前关卡数提升），两界都算，可无限重复；
     · 首次达到「新纪录关」→ 一次性大额（同关重复刷只拿微量）—— 即「首次更多」。
   刻意不做硬 cap：这是长线养成资源，靠神位成本曲线（1.010^lv）控制节奏，
   而不是靠卡死获取上限。

   消耗（供奉神位）：
     · 百分比类神位：上限 Lv.1000（用户要求：除特殊属性外一律 1000 级）；
     · 特殊属性（多杀上限 / 斩杀线）：整数型、收益陡峭，故独立设较低上限。

   只作用于凡人界：香火加成不参与任何灵界计算，不回流，避免双向循环。
   ══════════════════════════════════════════════════════════════ */
const INCENSE_CAP_PCT = 1000;      /* 百分比类神位等级上限 */
/* 供奉成本：分段递增，三段增长率逐级抬高。
   历史：① 全程 1.010^lv（太线性，几百级就满）→ ② 1.035/1.050/1.070（前快后慢，但仍偏快）
         → ③ 现 1.055/1.075/1.100：整体大幅拉高难度，后期陡到基本停滞。
     Lv.0~99    ×1.055/级   起量段：仍保留「点几下就涨」的即时反馈（约到 Lv.100 花 2.6e4）
     Lv.100~399 ×1.075/级   明显变陡，中期主要区间
     Lv.400~999 ×1.100/级   极陡：Lv.400 单级已达 1.1e13，累计 1.5e14；
                            约 Lv.470 起单级成本超过 2^53，统一夹到 SAFE_INT
   SAFE_INT 夹取只影响本就不可达的深水区，对该区间玩家无差别，仅防止 Infinity / NaN
   把购买判定搞坏。刻意不指望在当前获取途径下升满 —— 后续会新增香火获取方式。
   段与段之间连续衔接：边界只差一个增长率因子，不跳变也不回退（已由单调性测试覆盖）。 */
const INCENSE_COST_SEG = [
  {at:0,   g:1.12},      /* 原 1.055 → 1.08 → 1.12：持续提高灵界系统资源消耗 */
  {at:100, g:1.16},      /* 原 1.075 → 1.12 → 1.16 */
  {at:400, g:1.22}       /* 原 1.100 → 1.18 → 1.22 */
];
/* 特殊属性（疾神·多杀上限 / 斩神·斩杀线）成本额外 ×10。
   理由：它们是整数型、每级收益陡峭（多杀上限 +1 只、斩杀线 +0.4%），
   与「百分比类升 1 级只加 0.5%」完全不是一个量级，必须单独加价，
   否则玩家会无脑先把两个特殊神位拉满。此系数与 base 分离，便于单独调整。 */
const INCENSE_SP_COST_MUL = 10;
const INCENSE_SEATS = [
  {k:'atk',   n:'战 神', d:'凡人界 攻击',         kind:'pct', per:0.5, base:20},
  {k:'crit',  n:'锐 神', d:'凡人界 暴击 / 暴伤',  kind:'pct', per:0.4, base:25},
  {k:'spd',   n:'速 神', d:'凡人界 攻速',         kind:'pct', per:0.3, base:30},
  {k:'fdm',   n:'伤 神', d:'凡人界 最终伤害',     kind:'pct', per:0.5, base:25},
  {k:'psn',   n:'毒 神', d:'凡人界 毒伤害',       kind:'pct', per:0.5, base:25},
  {k:'mer',   n:'经 神', d:'经脉全部效果',        kind:'pct', per:1,   base:15},
  {k:'tower', n:'塔 神', d:'双塔掉落',            kind:'pct', per:1,   base:15},
  {k:'box',   n:'箱 神', d:'凡尘宝箱产出',        kind:'pct', per:1,   base:15},
  {k:'gf',    n:'功 神', d:'功法全部效果',        kind:'pct', per:1,   base:15},
  {k:'rb',    n:'转 神', d:'转生点获取',          kind:'pct', per:1,   base:15},
  /* —— 特殊属性：整数型、收益陡峭，独立上限，不参与 Lv.1000 —— */
  {k:'multi', n:'疾 神', d:'多杀上限 +1 / 级（需伤害溢出）', kind:'sp', max:100, base:200},
  {k:'exec',  n:'斩 神', d:'斩杀线 +0.4% / 级',              kind:'sp', max:50,  base:300}
];
function incenseDef(k){ for(let i=0;i<INCENSE_SEATS.length;i++) if(INCENSE_SEATS[i].k===k) return INCENSE_SEATS[i]; return null; }
function incenseMaxLv(k){ const d = incenseDef(k); if(!d) return 0; return d.kind === 'sp' ? d.max : INCENSE_CAP_PCT; }
function incenseLv(k){
  if(!S.incenseLv) S.incenseLv = {};
  const mx = incenseMaxLv(k);
  let v = Math.floor(Number(S.incenseLv[k]) || 0);
  if(!isFinite(v) || v < 0) v = 0;
  if(v > mx){ v = mx; S.incenseLv[k] = v; }
  return v;
}
function incenseCost(k){ return incenseCost2(k, incenseLv(k)); }
/* 百分比类神位的加成值（特殊属性不走这里） */
function incensePct(k){
  const d = incenseDef(k); if(!d || d.kind !== 'pct') return 0;
  return d.per * incenseLv(k);
}
/* 香火余额（大数） */
function incenseNum(){ return _toD(S.incense || D(0)); }
/* 一次最多能买多少级（用于「MAX」按钮）。
   上限取该神位的等级上限而非固定 200 —— 深水区成本陡增后，玩家攒够香火时
   200 级会明显不够用（要反复点 MAX）；而这里是纯算术循环，跑到满级也只有
   千次级 pow，开销可忽略，不会卡死。 */
function incenseAfford(k){
  const mx = incenseMaxLv(k);
  let n = 0, c = incenseCost(k);
  if(c === Infinity) return 0;
  let have = Number(dNum(incenseNum()));
  if(!isFinite(have)) return 0;
  while(n < mx && c !== Infinity){
    if(have < c) break;
    have -= c; n++;
    c = incenseCost2(k, incenseLv(k) + n);
  }
  return n;
}
/* 指定等级的供奉成本。分段累乘：先把前面各段整段乘完，再乘当前段的零头，
   保证段边界处连续（Lv.99→100 只差一个 ×1.035 与 ×1.050 的因子，不跳变）。 */
function incenseCost2(k, lv){
  const d = incenseDef(k); if(!d) return Infinity;
  const mx = incenseMaxLv(k);
  if(lv >= mx) return Infinity;
  /* 幂函数：base × (lv+1)^k（原为分段指数，后期会撞 SAFE_INT 被夹死在 9.01Qa） */
  let c = d.base * LMG_COST.inc.mul * (d.kind === 'sp' ? INCENSE_SP_COST_MUL : 1) * Math.pow(lv + 1, LMG_COST.inc.k);
  if(!isFinite(c) || c <= 0) return SAFE_INT;
  /* 后期成本远超 2^53，普通 Number 会丢精度 —— 这里只作为「数量级门槛」使用，
     故统一夹到 SAFE_INT，避免 Infinity / NaN 把购买判定搞坏。 */
  return Math.max(1, Math.min(SAFE_INT, Math.round(c)));
}
function incenseBuy(k, n){
  const d = incenseDef(k); if(!d) return;
  const mx = incenseMaxLv(k);
  let lv = incenseLv(k);
  if(lv >= mx){ toast('【' + d.n + '】已达上限 Lv.' + mx); return; }
  n = Math.max(1, Math.floor(n) || 1);
  let total = 0, buy = 0;
  for(let i=0;i<n && (lv+buy) < mx;i++){ total += incenseCost2(k, lv + buy); buy++; }
  if(buy <= 0) return;
  if(dcmp(incenseNum(), D(total)) < 0){
    toast('香火不足，需 ' + nStr(total) + '（当前 ' + dStr(incenseNum()) + '）'); return;
  }
  S.incense = dsub(incenseNum(), D(total));
  S.incenseLv[k] = lv + buy;
  toast('【' + d.n + '】Lv.' + (lv + buy) + (buy > 1 ? ('（+' + buy + '）') : '') + '　消耗香火 ' + nStr(total));
  logMsg('供奉【' + d.n + '】→ Lv.' + (lv + buy));
  renderIncense(); renderNow(true); save();
}

/* ---------- 多杀 / 斩杀线：特殊属性出口 ----------
   多杀：只有「本次攻击的伤害确实溢出、打穿了当前怪物」才会继续斩杀后续怪物。
   下一只怪血量按 BASE_GROW(1.062)^关卡 递增，因此能连斩几只完全由伤害溢出量决定，
   碾压越深连斩越多、卡关时自动退化为 1 只 —— 天然收敛，不需要额外门槛。 */
function multiKillCap(){ return 2 + incenseLv('multi'); }
function incenseExecLine(){ return incenseLv('exec') * 0.4; }

/* ---------- 获取：击杀（可重复，无上限）+ 首次新纪录更多 ---------- */
function incenseOnKill(isNew){
  const lv = (S.realm === 'lmg') ? (S.lmgLevel || 1) : (S.level || 1);
  let g = 1 + Math.floor(lv * 0.02);          /* 重复刷关：微量，无上限 */
  if(isNew) g += Math.floor(lv * 3);          /* 首次新纪录：明显更多 */
  S.incense = dadd(incenseNum(), D(g));
  if(isNew) S.incenseFirst = (S.incenseFirst || 0) + 1;
  return g;
}

/* ---------- 面板 ---------- */
function openIncense(){
  /* 顺序不能反：renderIncense() 有「面板未显示则 return」的守卫（挡掉关着时的无谓重绘），
     先 render 再 show 会导致首次打开内容为空。 */
  $('incenseOverlay').classList.add('show'); renderIncense();
}
function closeIncense(){ const o = $('incenseOverlay'); if(o) o.classList.remove('show'); }
function renderIncense(){
  const ov = $('incenseOverlay'); if(!ov || !ov.classList.contains('show')) return;
  const v = $('incenseVal'); if(v) v.textContent = dStr(incenseNum());
  const box = $('incenseSeatList'); if(!box) return;
  const have = incenseNum();
  let h = '';
  INCENSE_SEATS.forEach(function(d){
    const lv = incenseLv(d.k), mx = incenseMaxLv(d.k), maxed = (lv >= mx);
    const c1 = incenseCost(d.k);
    const c10 = incenseCost2(d.k, lv) * 0; let c10s = 0;
    for(let i=0;i<10 && (lv+i)<mx;i++) c10s += incenseCost2(d.k, lv + i);
    const aff = incenseAfford(d.k);
    const val = (d.kind === 'sp')
      ? (d.k === 'multi' ? ('多杀上限 ' + multiKillCap() + ' 只') : ('斩杀线 ' + (incenseExecLine()).toFixed(1) + '%'))
      : (d.k === 'tower'
          ? ('凡人塔转生点 +' + incensePct(d.k) + '% · 神兵塔碎片 +' + incensePct(d.k) + '%')   /* 双塔同时加成，分别列出 */
          : ('+' + incensePct(d.k) + '%'));
    const btn = function(n, cost, label){
      const ok = !maxed && cost > 0 && dcmp(have, D(cost)) >= 0;
      return '<div class="fs-btn ' + (ok ? 'b-gold' : 'b-brown') + '" data-cost="' + (maxed ? -1 : cost) + '" onclick="' + (maxed ? '' : ('incenseBuy(\'' + d.k + '\',' + n + ')')) + '"><span>' + label + '</span></div>';
    };
    h += '<div class="ic-seat">'
      +  '<div class="ic-sh"><span class="ic-sn">' + d.n + '</span>'
      +  '<span class="ic-sl">Lv.' + lv + ' / ' + mx + '</span></div>'
      +  '<div class="ic-sd">' + d.d + '　<b>' + val + '</b>'
      +  (d.kind === 'pct' ? ('　每级 +' + d.per + '%') : '') + '</div>'
      +  '<div class="ic-sb">'
      +  btn(1, c1, maxed ? '已 满 级' : ('+1 · ' + nStr(c1)))
      +  (maxed ? '' : btn(10, c10s, '+10 · ' + nStr(c10s)))
      +  (maxed ? '' : btn(Math.max(1, aff), (aff > 0 ? (function(){ let t=0; for(let i=0;i<aff;i++) t += incenseCost2(d.k, lv+i); return t; })() : 0), 'MAX'))
      +  '</div></div>';
  });
  box.innerHTML = h;
  const sm = $('incenseSum');
  if(sm){
    let tot = 0, maxed = 0;
    INCENSE_SEATS.forEach(function(d){ tot += incenseLv(d.k); if(incenseLv(d.k) >= incenseMaxLv(d.k)) maxed++; });
    sm.innerHTML = '神位总等级 <b class="hi">' + tot + '</b>　已圆满 <b class="hi">' + maxed + '</b> / ' + INCENSE_SEATS.length
      + '<br><span style="font-size:11px">击杀积累香火（无上限）· 仅凡人界</span>';
  }
}
/* 轻量刷新：只更新香火数与按钮可用态，不重建 DOM（render 每帧调用） */
function incenseRefresh(){
  const ov = $('incenseOverlay'); if(!ov || !ov.classList.contains('show')) return;
  const v = $('incenseVal'); if(v) v.textContent = dStr(incenseNum());
  const box = $('incenseSeatList'); if(!box) return;
  const have = incenseNum();
  Array.prototype.forEach.call(box.querySelectorAll('[data-cost]'), function(e){
    const c = Number(e.getAttribute('data-cost'));
    if(c < 0) return;
    const ok = dcmp(have, D(c)) >= 0;
    e.classList.toggle('b-gold', ok);
    e.classList.toggle('b-brown', !ok);
  });
}

/* ══════════════════════════════════════════════════════════════
   幽冥魂殿：灵魂的唯一消费入口（只在灵界开放）
   · 魂契 —— 灵魂买永久属性成长，是平行于「灵力升级」「修真」的第三条成长线
   · 魂市 —— 灵魂按比例兑换灵界资源（灵石 / 灵力 / 经验 / 洗练石 / 心魔票）
   · 魂召 —— 灵魂抽取灵兽碎片（与灵石抽取共用碎片池，但权重更优）

   为什么必须存在：灵魂（S.soul）此前只在「修真突破」一处消耗，主线/秘境/塔/
   BOSS/离线/洞府/分身七条产线全在产，玩家攒到天文数字却无处可花，资源等于装饰。
   这里一次性把出口铺满，且所有收益严格限定在灵界 —— 凡人界公式一行不动。
   ══════════════════════════════════════════════════════════════ */
const SOUL_PACT_MAX  = 40;      /* 每条魂契等级上限 */
/* 成本曲线：1.28^lv。单条全满约 2e7、10 条合计约 2e8 灵魂 —— 刻意压在「修真」（全破约 1e15）
   之下好几个数量级，让魂契成为「前期就能点、中期能铺满」的养成线，修真仍是长线目标；
   两者争夺同一份灵魂，形成取舍而非互斥。
   效果本身严格线性（每级固定增量），绝不做复利 —— 与全局「反指数」基调一致。 */
const SOUL_PACT_GROW = 1.85;   /* 原 1.28 → 1.55 → 1.85：持续提高灵界系统资源消耗 */
/* k 与 LMG_ATTR 一一对应，由 soulPactSum / soulPactPct / soulPactFix 三种口径接入 lmgAttrD()：
     exp → 百分比乘区（atk / hp / def / spd）      per 为「每级 +N%」
     pct → 百分点直加（crit / cdm / gp / gs / gu）  per 为「每级 +N 个百分点」
     fix → 点数直加（pen）                          per 为「每级 +N 点」
   base = 0 级升 1 级的灵魂成本；战斗属性贵、资源属性便宜，逼玩家做取舍。 */
/* 1.66：exp 类（攻/血/防/速）由「并入加法池」改为「独立乘区」。
   旧口径下 per 是相对属性 base 的百分比 —— 血量 base 仅 1000，而加法池总量常达数千 %，
   于是面板写着「灵界血量 +30%」，玩家按当前血量理解（以为 +1.9 万），实际只涨 1000×30%=300 点。
   改乘区后「+N%」= 当前值的 N%，名副其实；per 同步下调（40 级满级约 ×3.4 / ×3.0 / ×1.6）。
    1.67：再由「各自单独相乘」改为与职业同进「统一加成桶」—— 两系统叠乘的交叉项被消除，
    以后新增乘区系统只做线性相加、不会指数爆炸；桶整体另乘 LMG_ZONE_SCALE 削弱。 */
/* 1.68：per 已就地下调 ×0.7（取代全局 ZONE_SCALE），每条可单独微调 */
const SOUL_PACTS = [
  {k:'atk',   n:'噬魂锋', base:300, per:4,    kind:'exp', d:'灵界攻击'},   /* 6 → 4 */
  {k:'hp',    n:'冥王躯', base:260, per:4,    kind:'exp', d:'灵界血量'},   /* 6 → 4 */
  {k:'def',   n:'幽冥骨', base:240, per:3.5,  kind:'exp', d:'灵界防御'},   /* 5 → 3.5 */
  {k:'spd',   n:'疾魂步', base:400, per:1,    kind:'exp', d:'灵界攻速'},   /* 1.5 → 1 */
  {k:'crit',  n:'破妄眼', base:500, per:0.3,  kind:'pct', d:'灵界暴击'},   /* 0.4 → 0.3 */
  {k:'cdm',   n:'裂魂爪', base:320, per:18,   kind:'pct', d:'灵界暴伤'},   /* 25 → 18 */
  {k:'pen',   n:'穿魂刺', base:280, per:8,    kind:'fix', d:'灵界穿透'},   /* 12 → 8 */
  {k:'gp',    n:'汲灵诀', base:200, per:10,   kind:'pct', d:'灵力汲取'},   /* 1.75：14 → 10 */
  {k:'gs',    n:'寻魄术', base:200, per:10,   kind:'pct', d:'灵石寻获'},   /* 1.75：14 → 10 */
  {k:'gu',    n:'悟魂篇', base:220, per:13,   kind:'pct', d:'灵魂感悟'}    /* 1.75：18 → 13 */
];
function soulPactDef(k){ for(let i=0;i<SOUL_PACTS.length;i++) if(SOUL_PACTS[i].k===k) return SOUL_PACTS[i]; return null; }
function soulPactLv(k){
  if(!S.soulPact) S.soulPact = {};
  let v = Math.floor(Number(S.soulPact[k]) || 0);
  if(!isFinite(v) || v < 0) v = 0;
  if(v > SOUL_PACT_MAX){ v = SOUL_PACT_MAX; S.soulPact[k] = v; }   /* 惰性收敛：改档超限自动压回 */
  return v;
}
/* 升到下一级的灵魂成本；已满级返回 Infinity（调用方据此判断） */
function soulPactCost(k){
  const d = soulPactDef(k); if(!d) return Infinity;
  const lv = soulPactLv(k);
  if(lv >= SOUL_PACT_MAX) return Infinity;
  return Math.min(SAFE_INT, Math.round(soulPactCostNum(d.base * LMG_COST.pact.mul, lv)));
}
/* 幂函数成本的 Number 版（魂契用普通数字结算） */
function soulPactCostNum(base, lv){
  const f = base * Math.pow(lv + 1, LMG_COST.pact.k);
  if(!isFinite(f) || f <= 0) return SAFE_INT;
  return f;
}
/* ---- 三种口径的加成读取（供 lmgAttrD 接入）---- */
function soulPactSum(k){ const d = soulPactDef(k); return (d && d.kind === 'exp') ? d.per * soulPactLv(k) : 0; }
function soulPactPct(k){ const d = soulPactDef(k); return (d && d.kind === 'pct') ? d.per * soulPactLv(k) : 0; }
function soulPactFix(k){ const d = soulPactDef(k); return (d && d.kind === 'fix') ? d.per * soulPactLv(k) : 0; }
/* 魂契已投入的灵魂总量（面板统计用，不是可退资源） */
function soulPactSpent(){
  let s = 0;
  for(let i=0;i<SOUL_PACTS.length;i++){
    const d = SOUL_PACTS[i], lv = soulPactLv(d.k);
    for(let j=0;j<lv;j++) s += Math.min(SAFE_INT, Math.round(d.base * Math.pow(SOUL_PACT_GROW, j)));
  }
  return s;
}
/* 1.66：魂契预览 —— 「当前值 → 再升 1 级后的值」。
   玩家真正关心的是「我的血从多少变多少」，而不是一个会被加法池稀释的百分数。 */
function soulPactPreview(k){
  const d = soulPactDef(k);
  if(!d || d.kind !== 'exp') return null;
  if(soulPactLv(k) >= SOUL_PACT_MAX) return null;
  /* 1.67：魂契已进「统一加成桶」，且与职业共享同一个桶 —— 预览必须按桶口径算，
     不能再用旧的 (100+sum+per)/(100+sum)（那是各自单独相乘时的公式，会高估）。
     桶倍率 = 1 + (职业% + 魂契%) × LMG_ZONE_SCALE / 100 */
  const zone = clsPct(k) + soulPactSum(k);
  const m0 = 1 + zone * LMG_ZONE_SCALE / 100;
  const m1 = 1 + (zone + d.per) * LMG_ZONE_SCALE / 100;
  const now = _toD(lmgAttrD(k));
  const nxt = dmul(now, D(m1 / m0));
  return { now: now, next: nxt };
}
function soulPactBuy(k){

  const d = soulPactDef(k); if(!d) return;

  S.soulPact[k] = lv + 1;

/* ===== 1.48：魂市单价递增 =====
   原为固定单价，灵魂一多就能无限换，资源通胀失去稀缺性。
   改为「每买 1 份，下一份单价 ×1.08」，累计兑换越多越贵，天然形成软上限；
   同时把基准单价整体上调、单份产出下调（原 150 灵魂换 300 灵石明显偏高）。
   SOUL_MKT_MAXMUL 封顶：不封的话买满几百份后 Math.pow 会溢出成 Infinity。 */
const SOUL_MKT_GROW   = 1.08;
const SOUL_MKT_MAXMUL = 1e6;
function soulMarketBought(k){ return (S.soulBuy && S.soulBuy[k]) || 0; }
function soulMarketUnit(it, idx){
  const mul = Math.min(SOUL_MKT_MAXMUL, Math.pow(SOUL_MKT_GROW, Math.max(0, idx || 0)));
  return Math.round(it.cost * mul);
}
/* 一次买 n 份的总价：逐份累加（几何级数），不是「单价 ×n」——
   后者会让连买 100 份只按第 1 份计价，递增形同虚设。 */
function soulMarketCost(it, n, bought){
  let t = 0;
  for(let i = 0; i < n; i++) t += soulMarketUnit(it, (bought || 0) + i);
  return t;
}
function soulMarketDef(k){ for(let i=0;i<SOUL_MARKET.length;i++) if(SOUL_MARKET[i].k===k) return SOUL_MARKET[i]; return null; }
/* 单次「份」的产出量：随灵界当前关线性放大（第 1 关为基准量） */
function soulMarketAmt(it){
  const lv = Math.max(1, S.lmgLevel || 1);
  return it.scale ? Math.max(1, Math.round(it.amt * (1 + lv))) : it.amt;
}

  } else { resAdd('soul', cost); return; }  /* 未知项：不扣费 */
  S.soulBuy = S.soulBuy || {}; S.soulBuy[k] = (S.soulBuy[k] || 0) + n;
  logMsg('幽冥魂市：灵魂 -' + nStr(cost) + ' → ' + it.n + ' +' + nStr(amt));

/* 分身后台转生：只做数值结算。
   不能复用 doRebirth() —— 它开头就调 exitLingToFr()「转生属凡人界行为，先返回凡人界」，
   分身本就是为了「主角留在灵界、凡人界不停摆」，一转生就把玩家踢出灵界，与设计完全相悖。 */
function cloneDoRebirth(){
  const need = rebirthNeed();
  if((S.level || 1) < need) return false;
  const gain = rebirthGain();
  S.pts = dadd(S.pts || D(0), gain);
  S.rebirth = (S.rebirth || 0) + 1;
  if(typeof questAdd === 'function') questAdd('rb', 1);
  resAdd('tstone', 1);                   /* 与 doRebirth 同口径：每次转生 1 枚天赋石 */
  S.level = 1;
  S.poison = 0; S.ndxStack = 0;
  logMsg('分身代为转生：+' + dStr(gain) + ' 转生币，凡人界回归第 1 关（下次需第 ' + rebirthNeed() + ' 关）');
  if(typeof renderNow === 'function') renderNow(true);
  save();
  return true;
}
/* 分身每 tick 时执行凡人界自动化：先推关，达标再转生 */
function cloneAutoStep(){
  if(!frCloneUnlocked()) return;
  if(cloneTargetSecret()) return;              /* 1.56：挂机目标为秘境时，不再代管凡人界关卡 */
  if(!cloneAutoNextOn() && !cloneAutoRbOn()) return;
  /* 分身在凡人界代练，因此这里操控的是凡人界关卡 —— 与灵界无关 */
  if(cloneAutoNextOn()){
    const before = S.level || 1;
    /* 与离线推关同一口径：速率按历史最高关比例算，上限 stLv。
       传 byClone=true，否则「主角在灵界」的判断会让分身永远空转。 */
    const r = offlineFrPush(FR_CLONE_TICK_SEC / 3600, true);
    if(r && (S.level || 1) > before){ logMsg('分身推进凡人界：第 ' + r.from + ' 关 → 第 ' + r.to + ' 关'); }
  }
  if(cloneAutoRbOn()){
    const need = rebirthNeed();
    if((S.level || 1) >= need && cloneDoRebirth()){ toast('分身代为转生（第 ' + need + ' 关）'); }
  }
}

   玩家看到「击杀 0/时」会以为分身没生效，实际是【当前关卡怪物太硬、打不动】。 */

  if(!(k > 0)) return '<span style="color:#c98a8a">0（打不动）</span>';
  if(k >= 1) return nStr(Math.floor(k));
  return '<span style="color:#c98a8a">' + k.toExponential(1) + '</span>';
}

/* ==================== 凡人界分身 ====================
   解锁：100 次转生 + 已启灵（与启灵同一道门槛，属后期内容）。
   规则：只有主角在灵界时才产出；主角回到凡人界打怪则完全不产出 ——
        分身的意义是「去灵界后凡人界不至于停摆」，而不是让人挂着分身白嫖。
   产出：按凡人界当前关卡的击杀效率估算（DPS 与攻速 → 每小时击杀数），
        再乘以自己打怪的每击杀产出，最后统一打 FR_CLONE_RATE 的折扣 ——
        刻意低于自己亲手刷，亲自打永远更划算。
   无离线：分身只在在线 tick 中结算，不参与离线时长计算。 */
const FR_CLONE_RATE     = 0.35;      /* 分身效率：自己打怪的 35% */
const FR_CLONE_NEED_REB = 100;       /* 解锁所需转生次数 */
const FR_CLONE_TICK_SEC = 300;       /* 每 5 分钟结算一次 */
function frCloneUnlocked(){ return (S.rebirth || 0) >= FR_CLONE_NEED_REB && !!S.lmgUnlocked; }
/* 每小时击杀数：与灵界离线同口径（DPS / 怪物血量），但用凡人界的 calcDps + 当前关血量 */
function frCloneKillsPH(){
  try{
    const dps = calcDps();
    if(!dps || dcmp(dps, D(0)) <= 0) return 0;
    const hp = dmul(BASE_HP, dpow(BASE_GROW, Math.max(0, (S.level || 1) - 1)));
    const ld = Math.log10(Math.abs(dps.m || 0)) + Number(dps.e || 0n);
    const lh = Math.log10(Math.abs(hp.m || 0)) + Number(hp.e || 0n);
    const sec = Math.pow(10, lh - ld);
    if(!isFinite(sec) || sec <= 0) return 0;
    const aps = realAps();
    let k = 3600 / sec;
    if(!isFinite(k)) return 0;
    /* 1.56：接入香火「疾神·多杀」。
       单次攻击伤害 = DPS / 攻速，打穿当前怪后溢出伤害继续连斩（与主线 killBoss 同口径），
       故每次攻击最多斩 floor(DPS / (攻速 × 血量)) 只，再受疾神上限 multiKillCap() 约束。
       击杀/时 = 攻速 × 3600 × 每次攻击斩杀数，且不会超过理论上限 3600/sec（DPS ÷ 血量）。
       注：只有「伤害溢出」才触发多杀 —— 关卡高到打不动时仍为 1 只，多杀救不了，
       真正的解法是到转生线就转生（推关上限已卡在转生线）。 */
    let per = 1;
    if(aps > 0){
      const dmgPerHit = Math.pow(10, ld - Math.log10(aps));   /* 单次攻击伤害 */
      per = Math.max(1, Math.floor(dmgPerHit / Math.pow(10, lh)));
    }
    const cap = (typeof multiKillCap === 'function') ? multiKillCap() : 1;
    per = Math.min(cap, per);
    if(aps > 0) k = Math.min(k, aps * 3600 * per);    /* 攻速 × 多杀，且不超理论上限 */
    return Math.max(0, k) * FR_CLONE_RATE;
  }catch(e){ return 0; }
}
/* 给定 DPS 与怪物血量，返回「击杀/时」（含多杀上限与分身效率系数）。
   抽出为公共函数：凡人界关卡与秘境只是血量公式不同，效率算法完全一致。 */
function _cloneKphFor(dps, hp){
  const ld = Math.log10(Math.abs(dps.m || 0)) + Number(dps.e || 0n);
  const lh = Math.log10(Math.abs(hp.m || 0)) + Number(hp.e || 0n);
  const sec = Math.pow(10, lh - ld);
  if(!isFinite(sec) || sec <= 0) return 0;
  const aps = realAps();
  let k = 3600 / sec;
  if(!isFinite(k)) return 0;
  let per = 1;
  if(aps > 0){
    const dmgPerHit = Math.pow(10, ld - Math.log10(aps));
    per = Math.max(1, Math.floor(dmgPerHit / Math.pow(10, lh)));
  }
  const cap = (typeof multiKillCap === 'function') ? multiKillCap() : 1;
  per = Math.min(cap, per);
  if(aps > 0) k = Math.min(k, aps * 3600 * per);
  return Math.max(0, k) * FR_CLONE_RATE;
}
/* 分身在秘境的击杀效率（按当前秘境层数怪物血量） */
function frCloneSecretKillsPH(){
  try{
    const dps = calcDps();
    if(!dps || dcmp(dps, D(0)) <= 0) return 0;
    return _cloneKphFor(dps, secretMaxHpFor(Math.max(1, S.secretStage || 1)));
  }catch(e){ return 0; }
}
/* ── 1.91 内力加成：统一加法池，攻速单独乘 ──────────────────────────────
   原结构被拆成 4 个乘区（攻速 × 经脉矿脉 × BOSS × 加成池），区相乘：
   堆多个来源的实际收益是乘积而非加和，玩家看到 +200% 与 +600% 会以为相加，
   实测全额投入时 ×539（若真为单一加法池仅 ×15），内力因此膨胀失控。
   现合并：经脉 / 矿脉 / BOSS / 功法 / 遗物 / 宠物 / 宝物 / 称号 / 兵器 全部进同一加法池，
   只有攻速保留独立乘区（保留「堆攻速在秘境有额外回报」的设计意图）。
   口径与手动 secretKill 完全一致。 */
function secNeiliAddPct(){
  return merSum('neili') + maiFrDropPct('neili')          /* 足三阴经 + 矿脉 */
       + bossNeiliPct()                                    /* BOSS 通关 */
       + gfSum('neili') + relicVal('neili') + petSum('neili')
       + treSum('neili') + titleSum('neili') + weapNeili(); /* 功法/遗物/宠物/宝物/称号/兵器 */
}
function secretNeiliMul(){ return 1 + secNeiliAddPct() / 100; }

function frCloneSecretGainPH(){

     且怪物血量逐层递增；原实现按当前层速率整段外推，高估数十倍。 */
  return secSimHour(1, {rate: FR_CLONE_RATE, minSecPerKill: 1});
}
/* 分身秘境挂机：逐层模拟推进，产出与手动秘境同口径（含攻速加成）。
   秘境机制是「击杀 → 进下一层」，故分身会一路推到打不动为止，
   之后停在该层；若玩家开启了秘境自动重置，则由分身代为重置。 */
function cloneSecretStep(hrs){
  if(!frCloneUnlocked()) return null;
  if(S.realm !== 'lmg') return null;
  /* 玩家自己正在秘境里：分身不插手。
     否则分身推进 secretStage 会与手动 secretKill 抢同一份层数 / 血量，
     玩家刚打掉一半血的怪会被 secretInit() 换掉，表现为「血条莫名满血」。 */
  if(S.inSecret) return null;
  if(!(hrs > 0)) return null;
  let dps;
  try{ dps = calcDps(); }catch(e){ return null; }
  if(!dps || dcmp(dps, D(0)) <= 0) return null;
  const apsMul = secApsMul();
  const nlMul = secretNeiliMul();
  const startStage = Math.max(1, S.secretStage || 1);
  let stage = startStage;
  let kills = 0, nlSum = D(0), boxN = 0, soulN = 0;
  /* 每次 tick 最多推进 1 层，节奏与手动一致（最快一秒一层）。
     原实现把本次时间预算一次性算完：DPS 高时一个 tick 就连推几十层，
     层数飞涨、掉落日志刷屏，观感上完全不像「分身在代打」。
     改为用累积器攒够击杀所需时长才进一层，剩余时长留到下次 tick。 */
  _secCloneAcc = (_secCloneAcc || 0) + hrs;
  {
    const kph = _cloneKphFor(dps, secretMaxHpFor(stage));
    if(kph > 0){
      const need = 1 / kph;                       /* 杀掉当前层所需小时数 */
      if(_secCloneAcc >= need){
        _secCloneAcc -= need;
        kills += 1;
        let nl = secNeiliBase(stage);            /* 1.90：含重复层衰减 */
        nlSum = dadd(nlSum, dFloor(dmul(D(nl), D(apsMul * nlMul))));
        secMarkBest(stage);                      /* 1.90：分身推进也计入已击杀最高层 */
        soulN += 1;
        if(Math.random() < Math.min(0.95, SEC_BOX_BASE * apsMul)) boxN += 1;
        stage += 1;
      }
    }
  }
  if(kills <= 0){
    /* 完全打不动：仅在玩家开启了「秘境自动重置」时才代为重置，
       且冷却与玩家设定的秒数一致 —— 否则 clonePushTick 每秒一跑会每秒重置一次，
       日志刷屏且秘境层数永远卡在 1。 */
    if(secAutoOn()){
      const _now = Date.now(), _cd = secAutoSec() * 1000;
      if(!S.secCloneTs) S.secCloneTs = _now;                 /* 首次打不动：开始计时 */
      if(_now - S.secCloneTs >= _cd){
        S.secCloneTs = _now;
        resetSecret();
        logMsg('分身打不动，已自动重置');
      }
    }else{ S.secCloneTs = 0; }
    return null;
  }
  S.secCloneTs = 0;                                        /* 有击杀则清掉打不动计时 */
  S.secretStage = stage;
  S.secretKills = (S.secretKills || 0) + kills;
  S.stSecK = (S.stSecK || 0) + kills;
  if(dcmp(nlSum, D(0)) > 0)  S.neili = dadd(nlD(), nlSum);
  if(soulN > 0)  resAdd('soul', soulGain(soulN));
  if(boxN  > 0)  resAdd('box', boxN);
  /* 化肥 / 种子：期望值取整，零头进累加器，避免小数被抹 */
  const d = frCloneData();
  let fertRaw = (d.accFert || 0) + kills * FERT_DROP_BASE * apsMul;
  let fn = Math.floor(fertRaw); d.accFert = fertRaw - fn;
  for(let i = 0; i < Math.min(fn, 20); i++){
    const _f = fertRndOne(); fertAdd(_f.k, 1);
    logMsg('分身秘境掉落【' + _f.n + '】×1');
  }
  let seedRaw = (d.accSeed || 0) + kills * SEED_DROP_BASE * apsMul;
  let sn = Math.floor(seedRaw); d.accSeed = seedRaw - sn;
  for(let i = 0; i < Math.min(sn, 20); i++){
    const _si = seedRndOne(); seedAdd(_si, 1);
    logMsg('分身秘境掉落种子【' + SEEDS[_si].n + '】×1');
  }
  return {kills:kills, from:startStage, to:stage, neili:nlSum, soul:soulN, box:boxN};
}

function frCloneApply(hrs){
  if(!frCloneUnlocked()) return null;
  if(S.realm !== 'lmg') return null;              /* 只有主角在灵界才产出 */
  if(!(hrs > 0)) return null;
  const g = frCloneGainPH();
  if(!g) return null;
  const d = frCloneData();
  const r = {soul:0, box:0, gem:0, neili:0};
  /* 宝石 / 内力：整数累加，零头进 acc 防止小数被抹 */
  let gemRaw = (d.accGem || 0) + g.gem * hrs;
  r.gem = Math.floor(gemRaw); d.accGem = gemRaw - r.gem;
  let nlRaw = (d.accNl || 0) + g.neili * hrs;
  r.neili = Math.floor(nlRaw); d.accNl = nlRaw - r.neili;
  let boxRaw = (d.accBox || 0) + g.box * hrs;
  r.box = Math.floor(boxRaw); d.accBox = boxRaw - r.box;
  let soulRaw = (d.accSoul || 0) + g.soul * hrs;
  r.soul = Math.floor(soulRaw); d.accSoul = soulRaw - r.soul;
  if(r.gem > 0)   resAdd('gem', r.gem);
  if(r.neili > 0) nlAdd(r.neili);
  if(r.box > 0){  resAdd('box', r.box); }
  if(r.soul > 0)  resAdd('soul', soulGain(r.soul));
  return r;
}

}
/* 按权重强制抽一档（发放已累积的掉落数时用；与 fertRollDrop 的「先判概率」不同） */
function fertRndOne(){
  let tot = 0; FERT_W.forEach(function(x){ tot += x; });
  let r = Math.random() * tot;
  for(let i = 0; i < FERT.length; i++){ r -= FERT_W[i]; if(r <= 0) return FERT[i]; }
  return FERT[0];
}
function seedRndOne(){
  let tot = 0; SEED_W.forEach(function(x){ tot += x; });
  let r = Math.random() * tot;
  for(let i = 0; i < SEEDS.length; i++){ r -= SEED_W[i]; if(r <= 0) return i; }
  return 0;
}

/* 档位试算（1.99）：在不真实扣灵力的前提下，算出按当前档位能升几级、共花多少。
   与 maiPlan 同口径 —— 受「当前持有灵力」约束，故卡片上显示的数字就是点了真会花掉的数字。 */
function lmgUpPlan(k, n){
  const a = lmgAttrDef(k); if(!a) return {n:0, cost:D(0)};
  const _mx = lmgAttrMaxLv(k), lv0 = lmgAttrLv(k);
  const isMax = (n === 0 || n === 'max');
  const want = isMax ? ((_mx === null) ? Math.min(LMG_UP_MAX_BATCH, 2000) : LMG_UP_MAX_BATCH)
                     : Math.max(1, Math.floor(n) || 1);
  let have = _toD(S.lmgPower || D(0)), sum = D(0), lv = lv0, cnt = 0;
  for(let i = 0; i < want; i++){
    if(_mx !== null && lv >= _mx) break;
    const _s = S.lmgAttr[k]; S.lmgAttr[k] = lv;
    const c = lmgUpCost(k);
    S.lmgAttr[k] = _s;
    if(!c || dcmp(dadd(sum, c), have) > 0) break;
    sum = dadd(sum, c); lv++; cnt++;
  }
  return {n:cnt, cost:sum};
}
function lmgUpgrade(k){ lmgUpgradeN(k, 1); }
/* 批量升级（1.43）：×10 / MAX（n=0 表示 MAX）。
   不能简单 for 循环 n 次后统一扣款 —— 每级成本递增，必须逐档取价；
   且中途灵力不足时要「停在能负担的最后一级」，而不是整笔失败（否则 MAX 永远点不动）。
   ★ 关键坑：lmgUpCost 读的是 S.lmgAttr[k] 的当前等级，故必须先临时写入待升到的等级再取价，
     否则每次都取到同一档的价格（连升 10 级只扣 1 级的钱）。
   MAX 单次封顶 LMG_UP_MAX_BATCH 级，防止极端存档下循环过长卡住主线程。 */
const LMG_UP_MAX_BATCH = 100000;
function lmgUpgradeN(k, n){
  const a = lmgAttrDef(k); if(!a) return;
  const _mx = lmgAttrMaxLv(k);
  if(_mx !== null && lmgAttrLv(k) >= _mx){ toast(a.n + ' 已达等级上限 Lv.' + _mx); return; }
  const isMax = (n === 0);
  /* 1.82：资源三项已取消等级上限，若仍按 LMG_UP_MAX_BATCH(10万) 逐档试算会卡住主线程；
     无等级上限的属性，一次性批量压到 2000 级（纯性能保护，非设计上限，可连点 MAX 继续升）。 */
  const want = isMax ? ((_mx === null) ? Math.min(LMG_UP_MAX_BATCH, 2000) : LMG_UP_MAX_BATCH)
                    : Math.max(1, Math.floor(n) || 1);
  let pw = _toD(S.lmgPower || D(0)), done = 0, spent = D(0), stop = '';
  for(let i = 0; i < want; i++){
    const cur = lmgAttrLv(k) + done;
    if(_mx !== null && cur >= _mx){ stop = '已达上限'; break; }
    const _save = S.lmgAttr[k]; S.lmgAttr[k] = cur;
    const c = lmgUpCost(k);
    S.lmgAttr[k] = _save;
    if(!c || dcmp(pw, c) < 0){ stop = '灵力不足'; break; }
    pw = dsub(pw, c); spent = dadd(spent, c); done++;
  }
  if(done <= 0){ toast(a.n + '：' + (stop || '无法升级')); return; }
  S.lmgPower = pw;
  lmgAttrSetLv(k, lmgAttrLv(k) + done);
  if(k === 'hp') S.lmgHp = lmgAttrD('hp');
  logMsg(a.n + ' → Lv.' + lmgAttrLv(k) + '（连升 ' + done + ' 级，消耗 ' + nStr(spent) + ' 灵力）');
  toast(a.n + ' 连升 ' + done + ' 级' + (stop ? '（' + stop + '）' : ''));
  renderLing(); save();
}

/* 开启灵界宝箱：有概率出【一种】资源，数量五档随机（欧皇档 0.1%） */
function openLmgBox(all){
  if(!(S.lmgBox > 0)){ toast('没有灵界宝箱'); return; }
  const cnt = all ? Math.min(S.lmgBox, BOX_ALL_MAX) : 1;
  /* 五档权重累计 */
  const totW = LMG_BOX_TIERS.reduce(function(s,t){ return s + t.w; }, 0);
  let hitStone = 0, hitPower = 0, hitJade = 0, hitSoul = 0, hitLy = 0;
  const tierCnt = [0,0,0,0,0,0,0];
  for(let i = 0; i < cnt; i++){
    /* 等概率选一种资源 */
    const r = LMG_BOX_RES[Math.floor(Math.random() * LMG_BOX_RES.length)];
    /* 按权重选档位 */
    let rw = Math.random() * totW, tier = LMG_BOX_TIERS.length - 1;
    for(let j = 0; j < LMG_BOX_TIERS.length; j++){ rw -= LMG_BOX_TIERS[j].w; if(rw <= 0){ tier = j; break; } }
    tierCnt[tier]++;
    /* 该档数量 = 区间内随机整数 */
    const lo = r.amt[tier * 2], hi = r.amt[tier * 2 + 1];
    const got = lo + Math.floor(Math.random() * (hi - lo + 1));
    if(r.k === 'stone')      hitStone += got;
    else if(r.k === 'power') hitPower += got;
    else if(r.k === 'jade')  hitJade  += got;
    else if(r.k === 'soul')  hitSoul  += got;
    else                     hitLy    += got;
  }
  /* 扣减 */
  if(!isFinite(S.lmgBox) || S.lmgBox < 0) S.lmgBox = 0;
  if(S.lmgBox > SAFE_INT) S.lmgBox = SAFE_INT;
  S.lmgBox -= cnt; if(S.lmgBox < 0) S.lmgBox = 0;
  /* 发放（各资源口径与既有逻辑一致） */
  S.lmgStone = Math.min((S.lmgStone || 0) + hitStone, SAFE_INT);
  S.lmgPower = dadd(_toD(S.lmgPower || D(0)), D(hitPower));
  if(hitJade  > 0){ const _c = caveData(); _c.jade = Math.min(SAFE_INT, (_c.jade || 0) + hitJade); }
  if(hitSoul  > 0) resAdd('soul', hitSoul);
  if(hitLy    > 0){ const m = maiData(); m.ly = Math.min(SAFE_INT, m.ly + hitLy); }
  /* 提示 */
  let pt = '';
  if(hitStone > 0) pt += '　灵石 +' + nStr(hitStone);
  if(hitPower > 0) pt += '　灵力 +' + nStr(hitPower);
  if(hitJade  > 0) pt += '　洞府灵玉 +' + nStr(hitJade);
  if(hitSoul  > 0) pt += '　灵魂 +' + nStr(hitSoul);
  if(hitLy    > 0) pt += '　灵蕴 +' + nStr(hitLy);
  toast('开启灵界宝箱 ×' + nStr(cnt) + '：' + pt.trim());
  logMsg('开启灵界宝箱 ×' + nStr(cnt) + '：' + pt.trim());
  renderLing(); renderMai(); save();
}

         nStr(Math.floor(val)) + '/' + nStr(it.cap) + it.unit + over + '</i></span><b>+' + (Math.round(cur * 10) / 10) + '%</b></div>';
  });
  g.innerHTML = h;

/* 秘境模式下的单次攻击结算(被doHit调用) - 手动点击才闪白 */
function doSecretHit(manual){
  hitInc();
  if(!S.secretHp) secretInit();
    const _atk = calcAtk();                                /* 缓存攻击力（原每连击段重复计算，段数多时开销大） */
  let sum = dmul(_atk, D(0.9 + Math.random() * 0.2));
  let critN = 0, comboN = 0;
  const crit = Math.random() * 100 < calcCrit();
  if(crit){ sum = dmul(sum, D(critMult())); critN = 1 + critExtra();
    }
  sum = dadd(sum, dmul(_atk, D(calcIgn() / 100)));        /* 无视防御附加真伤（原三模式完全缺失） */
  { const _cs = {n:0, hit:false};
    sum = comboAdd(sum, _atk, 0.15, 0.9, 0.2, _cs);
    if(_cs.hit) comboN += _cs.n; }
  { const _pn = gfSum('psnhit');
    if(_pn > 0 && S.poison > 0) sum = dadd(sum, dmul(dmul(D(S.poison), D(_pn)), D(psnMultSum()))); }
  sum = dmul(sum, D(1 + tSum('nmult') / 100));            /* 猛攻之势 / 屠戮本能 */
  sum = dmul(sum, D(1 + calcFdm() / 100));                /* 最终伤害乘区 */
  if(ignAmpMul() > 1) sum = dmul(sum, D(ignAmpMul()));    /* 破界神锋 */
  S.ndxStack += 1 + tSum('ndxplus');                      /* 不死不休叠层（原三模式完全不叠） */
  sum = dmul(sum, D(1 + calcNdx() / 100));                /* 不死不休伤害加成 */
if(manual) S.clicks++;
  S.poison = Math.min(poisonCap(), S.poison + poisonAdd());
  S.totalDmg = dadd(S.totalDmg || D(0), sum);
  S.secretHp = dadd(S.secretHp, dNeg(sum));
  /* 防止负血量 */
  if(dcmp(S.secretHp, D(0)) < 0) S.secretHp = D(0);

  let poison = D(0);
  if(psnPerStack() > 0){ poison = calcPsnDmg(S.poison);
    S.totalDmg = dadd(S.totalDmg || D(0), poison);
    S.secretHp = dadd(S.secretHp, dNeg(poison));
    if(dcmp(S.secretHp, D(0)) < 0) S.secretHp = D(0);
  }
  achHit(sum, crit, poison);                           /* 移出毒伤分支 */

  /* 斩杀线天赋（原仅主线判定，秘境中失效） */
  const _line = tSum('line');
  if(_line > 0 && dcmp(S.secretHp, dmul(S.secretMaxHp, D(_line / 100))) <= 0 && dcmp(S.secretHp, D(0)) > 0) S.secretHp = D(0);
  if(dcmp(S.secretHp, D(0)) <= 0){
    secretKill();   /* 血量归零立即结算下一关（secretKill 会重置为下一关满血，天然防重入） */
  }
  return {sum:sum, crit:crit, critN:critN, combo:comboN>0, comboN:comboN, poison:poison};
}

/* 秘境自动攻击tick - 与手动完全一致（叠毒层+毒伤+击杀），不闪白 */
function secretTick(){
  if(!S.inSecret) return;
  /* 1.56 自动重置：超时未击杀则重置（先于攻击判定，避免白打一下） */
  if(secAutoOn() && S.secStageTs){
    if(Date.now() - S.secStageTs >= secAutoSec() * 1000){
      logMsg('秘境 ' + secAutoSec() + ' 秒打不死怪物，自动重置');
      toast('秘境自动重置（' + secAutoSec() + ' 秒打不死）');
      resetSecret();
      return;
    }
  }
  /* 2.03：按当前攻速批量结算本次 tick 的攻击次数（秘境不再被 fps 间隔锁死在 2~6 次/秒）。
     n = 攻速 × tick 间隔，至少 1 次；设 SEC_TICK_MAX 上限避免极高攻速时单帧结算过多导致卡顿。 */
  const _iv = (S.fps === 30 ? 500 : (S.fps === 120 ? 160 : 320));
  const _a  = (typeof calcAps === 'function') ? calcAps() : 1;
  let _n = Math.round(((isFinite(_a) && _a > 0) ? _a : 1) * _iv / 1000);
  if(!isFinite(_n) || _n < 1) _n = 1;
  if(_n > SEC_TICK_MAX) _n = SEC_TICK_MAX;
  let r = null;
  for(let _i = 0; _i < _n; _i++){
    if(!S.inSecret) break;                 /* 中途退出秘境（切界/自动重置）立即停止 */
    const _r = doSecretHit(false);
    if(_r) r = _r;
    if(_i < _n - 1) continue;              /* 只飘最后一次的字，避免单帧刷屏 */
  }
  if(r) popDmg(r, null, lmgDmgHost());
  renderSecretMode();
}

/* 1.56：秘境试炼按「当前攻速」加成爆率与内力。
   设计口径 —— 攻速越高，单位时间击杀越多，若只按「每次击杀」结算，
   高攻速玩家的秘境收益会被攻速上限（原 APS_CAP=50，1.57 起已取消）压平；
   故额外给一份随攻速增长的系数，让堆攻速在秘境里有直接回报。
/* 1.57：攻速取消封顶后，若仍按每点 +8% 线性，满配（约 1210 次/秒）会算出 ×97 —— 失控。
   改为分段：50 以内维持原 8%/点（手感与封顶时期完全一致），
   超出 50 的部分按 0.8%/点 计入 —— 仍然随攻速继续增长、不封顶，只是增速放缓。
   区间：攻速 1 → ×1.00；攻速 50 → ×4.92（与封顶时一致）；攻速 200 → ×6.2；攻速 1210 → ×14.3。 */
const SEC_APS_PER = 0.08;                    /* 每 1 点攻速 +8%（50 以内） */
const SEC_APS_PER_HI = 0.008;                /* 超出 50 的部分：每 1 点 +0.8%（不封顶） */
const SEC_APS_KNEE = 50;
/* 2.03：单个秘境 tick 内最多结算的攻击次数（防极高攻速时单帧卡顿）。
   标准档 320ms → 上限 200 次 ≈ 625 次/秒；极速档 160ms → 1250 次/秒，正常玩法远达不到。 */
const SEC_TICK_MAX = 200;
function secApsMul(){
  /* 2.03：秘境掉落不再吃攻速加成，恒返回 1。
     原设计是「攻速越高 → 每次击杀的掉落率越高」，而 2.03 起秘境已按攻速批量结算攻击次数
     （打得越快，击杀数本身就越多），掉落若再乘一次攻速就是双重放大 —— 故此处取消。
     保留函数与常量仅为兼容旧调用点，不再产生任何倍率。 */
  return 1;
}
/* ── 1.90 秘境重复层内力衰减 ────────────────────────────────────────────
   原机制：秘境每层内力随层数呈平方增长，而重置（手动 / 自动重置 / 分身代打重置）
   会把层数打回第 1 关 —— 同一批层数可以原路无限重刷，「卡层刷内力」成了最优解，
   继续推进新层反而没有收益优势，内力因此膨胀失控。
   现引入「已击杀过的最高层」secBest：
     · 层数 > secBest（首次打到的新层）→ 内力 100%，破新层的收益完全保留
     · 层数 ≤ secBest（重置后重刷的旧层）→ 内力 × SEC_REPEAT_RATE
   反复刷的收益被压到 1/N，首次推进与继续破层不受任何影响。
   注：判定口径是「已击杀」而非「已到达」，故卡住的那层下次打通时仍算新层（满额）。 */
/* 2.21：0.15 → 0.05。
   原 15% 仍足以让「卡层重刷」成为最优解：满配开自动重置时，1 小时可把 1~387 层整段扫 22 轮
   （实测 8514 击杀 / 65.26B 内力），而真实手动推进新层 1 小时只有约 29 击杀 / 3.04B。
   降到 5% 后重刷收益约为首通的 1/20，首次破层与继续推进不受任何影响。 */
const SEC_REPEAT_RATE = 0.05;                    /* 重复层内力倍率（0.05 = 5%） */
function secBest(){ return Math.max(0, S.secBest || 0); }
function secIsRepeat(stage){ return (stage || 0) <= secBest(); }
function secRepeatMul(stage){ return secIsRepeat(stage) ? SEC_REPEAT_RATE : 1; }
/* 单层内力基准：线性 ×6 + 二次项，首领关 ×5；重复层再乘 SEC_REPEAT_RATE
   1.91：二次项系数 0.5 → 0.15（A 档温和曲线）；
   1.92：二次项改为 1.4 次幂（见 SEC_NL_POW 注释），修正后期换灵力过剩。
   1.93：幂次 1.4 → 1.2，后期仍偏多，继续压平（单层 1000 层 ×0.46、10000 层 ×0.27）。 */
/* 1.92：内力曲线由「二次」降为「1.4 次幂」—— 后期「内力换灵力过剩」的对症修正。
   根因：单层内力 ∝ s²，而灵界修炼成本 ∝ lv^1.18（powCost: base×(lv+1)^1.18）。
   秘境层数随挂机时间线性推进（每 tick +1 层），故内力累计 ∝ t³、灵力需求 ∝ lv^2.18 ——
   产出增速（层数翻倍 → ×4）远超消耗增速（等级翻倍 → ×2.26），
   越到后期兑换越过剩（每翻倍过剩约 1.77 倍），这就是「后面感觉太多」的来源。
   改 1.4 次幂后：层数翻倍 → 内力 ×2.64，与成本 ×2.26 基本持平（过剩降到约 1.17 倍/翻倍）。
   数值影响（单层基准）：1000 层 ×0.27、5000 层 ×0.08、10000 层 ×0.053
   —— 前期几乎不变，层数越深削得越多，正对「后期过剩」。
   1.93：幂次再降 1.4 → 1.2。层数翻倍内力 ×2.64 → ×2.30，与成本 ×2.26 几乎持平，
   过剩速率由约 1.17 倍/翻倍降到约 1.02 倍/翻倍，后期不再滚雪球。
   数值影响（相对 1.92，单层基准）：100 层 0.70、1000 层 0.46、5000 层 0.32、10000 层 0.27。 */
const SEC_NL_POW  = 1.2;                     /* 内力随层数的幂次：调大=更陡（2.0 即恢复平方）
                                                  1.93：1.4 → 1.2，后期内力换灵力仍偏多，继续压平 */
const SEC_NL_COEF = 1.2;                       /* 幂项系数：调大即整体提高内力（2.02：2 → 1.2） */
/* 2.02：内力整体缩放（作用于线性项 + 幂项之和，调小即全线减少）。
   COEF 只影响幂项，而线性项 (12+2s)×6 在中后期占比很大，单改 COEF 削不动 ——
   故加一个总闸，与 COEF 配合：相对 2.01 单层内力约降至 0.59（-41%）。 */
const SEC_NL_SCALE = 0.35;                      /* 2.04：0.7 → 0.35（基础内力砍半） */
function secNeiliPowTerm(stage){
  var s = Math.max(0, stage || 0);
  if(s <= 0) return 0;
  var v = SEC_NL_COEF * Math.pow(s, SEC_NL_POW);
  if(!isFinite(v) || v > 9e15) v = 9e15;
  return v;
}
function secNeiliBaseRaw(stage){
  var nl = Math.floor(((12 + stage*2) * 6 + secNeiliPowTerm(stage)) * SEC_NL_SCALE);
  if(stage % 10 === 0) nl = Math.floor(nl*5);
  return nl;
}
function secNeiliBase(stage){
  var nl = secNeiliBaseRaw(stage);
  if(secIsRepeat(stage)) nl = Math.floor(nl * SEC_REPEAT_RATE);
  return nl;
}
function secMarkBest(stage){ if((stage || 0) > secBest()) S.secBest = stage; }
function secretKill(){
  S.ndxStack = 0;                                    /* 击杀后清空不死不休层数（主线 killBoss 已清，此处补齐） */
  var stage = S.secretStage;
  /* 单层内力基准（含首领 ×5、重复层衰减） */
  var nl = secNeiliBase(stage);
  /* 1.56：内力再乘秘境攻速加成 —— 高攻速在秘境获得直接回报 */
  nl = Math.floor(nl * secApsMul());
  if(!isFinite(nl) || nl > 9e15) nl = 9e15;               /* 防超双精度安全整数后扣减失真 */
  /* 1.91：攻速独立乘区 × 统一加法池（经脉/矿脉/BOSS/功法/遗物/宠物/宝物/称号/兵器 全部相加） */
  S.neili = dadd(nlD(), dFloor(dmul(D(nl), D(secretNeiliMul()))));
  S.secretKills++;
  resAdd('soul', soulGain());                            /* 灵魂：秘境击杀 +1 */
  S.stSecK = (S.stSecK || 0) + 1;
  /* 1.56：秘境极低概率掉落化肥（品质越高越难出），受攻速加成影响 */
  { const _fd = fertRollDrop(secApsMul());
    if(_fd){ fertAdd(_fd.k, 1); logMsg('秘境掉落【' + _fd.n + '】×1（成熟时间 -' + _fd.d.replace('所有地块成熟时间 -','') + '）'); } }
  /* 进阶丹资源已取消，境界突破改走宝石；首领关不再掉落任何进阶丹 */
  /* 秘境关卡进度不再写入日志 */
  questAdd('kill', 1);
  /* 1.56：宝箱爆率受攻速加成（原固定 18%） */
  if(Math.random() < Math.min(0.95, SEC_BOX_BASE * secApsMul())){ resAdd('box', 1); questAdd('getbox', 1); logMsg('秘境掉落凡尘宝箱 ×1'); }
  /* 1.56：秘境恢复种子掉落（原「不再掉落种子」注释已作废）—— 低概率，含 36/48/72 小时长时档 */
  { const _si = seedRollDrop(secApsMul());
    if(_si >= 0){ seedAdd(_si, 1); logMsg('秘境掉落种子【' + SEEDS[_si].n + '】×1'); } }
  S.poison = 0;                                          /* 每层换怪，毒层从 0 重新叠 */
  secMarkBest(stage);                                    /* 1.90：记录已击杀最高层（重复层判定基准） */
  S.secretStage++;
  secretInit();
  renderSecretMode(); save();
}

/* 进入秘境模式：主界面切换为秘境战斗 */
function enterSecret(){
  exitLingToFr();                     /* 同上：进入秘境前先返回凡人界 */
  cancelSecret();
  if(S.inTower){ exitTower(); }
  clearBossMode();                    /* 进入秘境时自动退出 BOSS 挑战 */
  secretInit();
  S.inSecret = true;
  S.poison = 0;
  /* 秘境结算只保留 secretTimer 一路：
     autoTimer 每次 tick 调 doHit → 会路由到 doSecretHit（第二路伤害），
     秘境节奏由 S.fps 独立控制（500/320/160ms），本就与攻速无关，停掉 autoTimer 不影响设计意图。 */
  stopAuto();
  applySecretUI();
  startSecret();
  renderSecAutoBar();
  renderSecretMode();
  logMsg("进入秘境! 当前第 " + S.secretStage + " 关");
  toast("进入秘境 第 " + S.secretStage + " 关");
  save();
}

/* 退出秘境模式：恢复主线 */
function exitSecret(){
  clearBossMode();                    /* 返回主线时自动退出 BOSS 挑战 */
  S.inSecret = false;
  stopSecret();
  S.secretHp = null;
  applySecretUI();
  startAuto();                            /* 退出秘境：恢复主线自动攻击 */
  render();
  logMsg("离开秘境，返回主线战斗");
  toast("返回主线战斗");
  save();
}

/* 重置秘境：回到第 1 关，但保持在秘境内（不再返回主线，无需重新进入） */
function resetSecret(){
  S.secretStage = 1;
  S.secretKills = 0;
  secretInit();                                    /* 重新生成第 1 关满血 */
  /* 玩家此刻不在秘境内（身处灵界 / 主线，或本次由分身后台触发）：只重置数据，一律不碰 UI。
     原实现无条件置 S.inSecret = true 再调 applySecretUI() / startSecret()，后果有三：
     ① syncEnemyVisual() 走 inSecret 分支 → 灵界背景冒出秘境狼图覆盖层（玩家看到的「串画面」）；
     ② startSecret() 启动 secretTimer → 玩家在灵界的攻击被 doHit 路由到 doSecretHit，打到秘境怪身上；
     ③ S.poison 被清零 → 抹掉玩家当前所在模式（灵界）的毒层。 */
  if(!S.inSecret) return;
  S.poison = 0;
  var bt = $("bossTag"); if(bt) bt.style.display = "none";   /* 第1关非首领关 */
  setEnemyImg('main');                             /* 还原普通怪物图 */
  applySecretUI();
  startSecret();                                   /* 重启秘境自动攻击 */
  renderSecAutoBar();
  renderSecretMode();
  toast("秘境已重置至第 1 关");
  logMsg("秘境重置，回到第 1 关");
  save();
}

/* 渲染秘境模式主界面 */
function renderSecretMode(){
  if(!S.inSecret) return;
  renderSecAutoBar();                                        /* 有签名缓存，高频调用无开销 */

  $("smStage").textContent = S.secretStage;
  const isBossStage = (S.secretStage % 10 === 0);          /* 每 10 关为首领关 */
  $("smBoss").textContent = "";                            /* 首领标识已移至血条上方，此处留空以杜绝换行跳动 */
  const bt = $("bossTag"); if(bt) bt.style.display = isBossStage ? "" : "none";
  setEnemyImg(isBossStage ? 'secret' : 'main');             /* 首领关切换专属立绘 */
  $("smNeili").textContent = nStr(S.neili);

  if(S.secretMaxHp && S.secretHp){
    var displayHp = dcmp(S.secretHp, D(0)) < 0 ? D(0) : S.secretHp;
    var ratio = (displayHp.m / S.secretMaxHp.m) * Math.pow(10, Number(displayHp.e - S.secretMaxHp.e));
    var pct = Math.max(0, Math.min(100, ratio*100));
    $("hpFill").style.width = pct + "%";
    $("hpText").textContent = dStr(displayHp) + " / " + dStr(S.secretMaxHp);
  }

}

      const v = o[k];
      if(v && typeof v === 'object' && v.m !== undefined) bak[k] = { __d:1, m:String(v.m), e:String(v.e) };
      /* 普通对象 / 数组只序列化一次后以字符串留存（__s 标记）。
         原写法 JSON.parse(JSON.stringify(v))：先 stringify 再 parse 还原成对象，
         随后整体 JSON.stringify(o) 又把它再序列化一遍 —— 同一份数据走了三遍，大存档明显浪费。
         改为以字符串形式落 _bak 后，最终整体序列化只需直接拷贝字符串。 */
      else if(v && typeof v === 'object'){ try{ bak[k] = { __s:1, j:JSON.stringify(v, bigRepl) }; }catch(e){ bak[k] = v; } }
      else bak[k] = v;
    }
    o._bak = bak;
    o._sig = sigOf(o);            /* 防篡改签名（导出时随档一起走，分享不受影响）；签名含 _bak */
    /* _bak 单独拼接，避免二次序列化：
       bak 内的对象已存成字符串（__s 标记），若随 o 整体 stringify 会被再转义一遍 ——
       大存档下等于把整份快照多走一次序列化，保存耗时翻倍。
       先摘掉 _bak 序列化主体，再把 bak 的 JSON 原文拼到末尾（键序不影响 parse）。 */
    let _json;
    try{
      const bakJson = JSON.stringify(bak, bigRepl);
      delete o._bak;
      const body = JSON.stringify(o, bigRepl);
      _json = body.slice(0, -1) + ',"_bak":' + bakJson + '}';
      o._bak = bak;               /* 还原，保持 o 完整（导出存档码等处可能复用同一对象） */
    }catch(_e){ _json = JSON.stringify(o, bigRepl); }
    localStorage.setItem('xiantu_save', _json);
    _saveFailed = 0;
  }catch(e){
    /* 配额耗尽（典型 5MB 上限）/ 无痕模式下 localStorage 写入会抛异常。
       原实现是空 catch：玩家会毫无察觉地持续丢进度（每 5 秒一次全丢），
       且没有任何线索可查。现首次失败即提示，并区分「配额满」与「存储不可用」。 */
    _saveFailed = (_saveFailed || 0) + 1;
    if(_saveFailed <= 3){
      const _q = !!(e && (e.name === 'QuotaExceededError' || e.name === 'NS_ERROR_DOM_QUOTA_REACHED' || e.code === 22 || e.code === 1014));
      try{ toast(_q ? '存储已满，进度无法保存！请先导出存档' : '无法写入本地存储（无痕模式？），进度不能保存'); }catch(_e){}
      try{ logMsg('⚠ 存档写入失败：' + (e && e.name || 'unknown') + '，进度未保存'); }catch(_e){}
    }
    if(window.console) console.warn('[save] 写入失败', e);
  }
}
/* ==================== 轻量防篡改（防随手改数字，不影响存档分享） ====================
   存档写入时附签名；读取时重算比对。
   - 自己导出/导入、分享给别人：签名随档走，完全正常
   - 用文本编辑器改数字：签名对不上 → 提示并修正，不清档
   - 老存档无签名：静默补签
   注：纯前端单机无法 100% 防破解，此为"防随手改"的轻量方案。
   ================================================================================ */
const SAVE_SIG_KEY = 'XT-SIG-2024';   /* 固定密钥：保证跨设备分享存档仍有效 */
/* ===== 签名覆盖范围：存档全部字段（排除签名自身） =====
   原实现仅覆盖 26 个「关键数值」字段，导致 talents / meridian / achClaim / gongfa / pets /
   treasures / relics / boss_* 等大量进度字段被改也检测不到 —— 防篡改形同虚设。
   现改为排除法：只跳过 _sig（签名自身），其余全部纳入摘要，任何字段被改都会破坏签名。
   ===== 序列化 helpers ===== */
const SIG_SKIP = new Set(['_sig']);                       /* 仅签名自身不参与 */
const BAK_SKIP = new Set(['_sig','_t','_bak','v']);       /* 回滚快照排除：签名 / 时间戳 / 自身 / 版本号 */
/* 稳定序列化：大数转 '#m e'、对象转有序 JSON、其余转字符串；保证同一存档每次结果一致 */
function serVal(v){
  if(v === null || v === undefined) return '\u0000';
  if(typeof v === 'object'){
    if(v.m !== undefined) return '#' + v.m + 'e' + v.e;
    /* 数组必须单独处理，不能混进对象分支：
       · 原分支用 Object.keys().sort() 对下标做「字典序」排序，元素数 ≥11 时顺序变成 0,1,10,11,…,2,3
         —— 虽然结果仍确定（签名不会误报），但长度越长越反直觉，且与下方新格式不兼容；
       · 更关键的是原格式把数组写成 {0:…,1:…}，与「键为 0/1 的普通对象」输出完全相同，
         即 ["a"] 与 {0:"a"} 签名一致 —— 攻击者可把数组替换成同键对象而不触发篡改检测，
         随后 tSum() 里的 indexOf() 会在运行时抛错，属于类型混淆隐患。
       现改为数值下标顺序 + 独立的 [ ] 包裹，与对象区分开。 */
    if(Array.isArray(v)){
      let a = '[';
      for(let i = 0; i < v.length; i++) a += serVal(v[i]) + ',';
      return a + ']';
    }
    try{
      const keys = Object.keys(v).sort();
      let r = '{';
      for(let i = 0; i < keys.length; i++) r += JSON.stringify(keys[i]) + ':' + serVal(v[keys[i]]) + ',';
      return r + '}';
    }catch(e){ return '{?}'; }
  }
  return String(v);
}
function sigOf(o){
  const keys = Object.keys(o).filter(function(k){ return !SIG_SKIP.has(k); }).sort();
  let str = '';
  for(let i = 0; i < keys.length; i++) str += keys[i] + '=' + serVal(o[keys[i]]) + ';';
  let h1 = 5381, h2 = 52711, h3 = 0x811c9dc5;
  const salt = SAVE_SIG_KEY;
  for(let i = 0; i < str.length; i++){
    const c = str.charCodeAt(i) + salt.charCodeAt(i % salt.length);
    h1 = ((h1 << 5) + h1 + c) >>> 0;
    h2 = (h2 + (h2 << 7) + c) >>> 0;
    h3 = ((h3 ^ c) * 16777619) >>> 0;
  }
  return h1.toString(36) + '-' + (h2 ^ h3).toString(36) + '-' + ((h1 + h2 + h3) >>> 0).toString(36);
}
/* 数值合理性：拦截负数 / NaN / 天文数字 */
/* 双精度安全整数上限：超过 2^53 后 n-1 与 n 无法区分，
   宝箱 / 宝石这类「普通 Number 存储且要做减法」的资源必须封顶在此，
   否则 S.box -= cnt 会静默失效（1e20-200000 仍是 1e20），「全部开启」可无限点。 */
const SAFE_INT = Number.MAX_SAFE_INTEGER;            /* 9007199254740991 */
/* 运行时资源累加封顶：SANE_MAX 只在读档时生效，运行时 S.gem += x / S.soul += n 从不封顶。
   gem 被改到 1e16 后超过 SAFE_INT，加减法开始静默失精（1e20 - 200000 仍是 1e20），
   「全部开启」这类依赖减法的功能会失效 —— S.box 那处已单独加过保护，其它资源没有。 */
/* 大数取整：仅对可精确表示的部分生效；超过 1e15 后小数位已无意义，原样返回 */
function dFloor(x){
  x = _toD(x);
  if(x.m === 0 || !isFinite(x.m)) return x;
  const e = Number(x.e);
  if(!isFinite(e)) return x;
  if(e < 0) return {m:0, e:0n};
  if(e > 15) return x;                                 /* 尾数不可辨，无需再取整 */
  return D(Math.floor(x.m * Math.pow(10, e)));
}
/* ---- 2.07 内力大数读写层 ----
   S.neili 由普通 Number 改为大数 {m,e}，与灵力 lmgPower 同口径，彻底解除 9.007e15 上限。
   原上限并非设计意图，而是 Number.MAX_SAFE_INTEGER 的精度天花板：超过后 1e16+1 === 1e16，
   加减法静默失效（刷到手的内力加不进去、兑换时扣不掉），capRes 只是在精度崩坏前夹住。
   旧档 neili 仍是普通数字，nlD() 自动兼容，无需迁移。 */
function nlD(){ const v = S.neili; return (v && typeof v === 'object' && v.m !== undefined) ? v : D(Number(v) || 0); }
function nlAdd(v){ S.neili = dadd(nlD(), _toD(v)); }
function nlHas(cost){ return dcmp(nlD(), _toD(cost)) >= 0; }
function nlPay(cost){ if(!nlHas(cost)) return false; S.neili = dsub(nlD(), _toD(cost)); return true; }

/* ---- 2.23 常规资源大数读写层 ----
   gem / box / soul / relicFrag / tstone 由普通 Number 改为大数 {m,e}，与 neili、lmgPower 同口径，
   彻底解除 9.007e15（Number.MAX_SAFE_INTEGER）上限。
   此前 capRes 只在读档时夹断，运行时 S.gem += x 从不封顶 —— 超过 SAFE_INT 后整数不再连续
   （1e16 + 1 === 1e16），加减法静默失效：「全部开启」这类依赖减法的功能会失效（扣不掉＝无限白嫖），
   刷到的资源也可能加不进去。旧档仍是普通数字，resD() 自动兼容，无需迁移。 */
const RES_BIG = ['gem', 'box', 'soul', 'relicFrag', 'tstone'];
function resD(k){ const v = S[k]; return (v && typeof v === 'object' && v.m !== undefined) ? v : D(Number(v) || 0); }
function resSet(k, v){ S[k] = _toD(v); }
function resAdd(k, v){ S[k] = dadd(resD(k), _toD(v)); }
function resSub(k, v){ S[k] = dsub(resD(k), _toD(v)); }
function resHas(k, c){ return dcmp(resD(k), _toD(c)) >= 0; }
function resPay(k, c){ if(!resHas(k, c)) return false; S[k] = dsub(resD(k), _toD(c)); return true; }
/* 仅在「确定量级远小于 1e15」的场合取 Number（如与 BOX_ALL_MAX 取 min）；量级过大返回 Infinity，语义仍正确 */
function resNum(k){ const d = resD(k); const e = Number(d.e); return (e > 15) ? Infinity : Number(d.m) * Math.pow(10, e); }
function capRes(k, v){
  const m = SANE_MAX[k];
  if(m === undefined || !isFinite(v)) return isFinite(v) ? v : 0;
  if(v > m) return m;
  if(v < 0) return 0;
  return v;
}

const SANE_MAX = { level:1e7, kills:1e12, rebirth:1e6,
                   farmLv:1e6, farmStage:100, farmOpen:10,
                   poison:1e12, ndxStack:1e9,
                   secretStage:1e7, towerFloor:1e7, towerDone:1e7, sbDone:1e7, drawCnt:1e9, bossWin:1e9,

     Math.min(poisonCap(), S.poison + poisonAdd())，轻则毒伤 NaN，重则累加出天文数字。 */
/* 注：gem 以普通 Number 存储，安全上限 9.007e15（SANE_MAX 的上限远大于该值，
   实测到该量级前会先触及浮点精度问题）。
   neili 自 2.07 起改为大数 {m,e}，已从本表移除 —— 保留会把读档值夹回 9.007e15，
   使大数改造形同无效（实测 1.2345e25 读档后变 9.01Qa）。 */
/* 数值合理性修正：拦截负数 / NaN / 天文数字。
   注意：修正后必须同步修正备份快照 _bak 中的同名字段，否则 sigOf() 会因主值与 _bak 不一致
        而判定为「篡改」并回滚——合法存档被误杀（如 level 被封顶到 1e7 而 _bak 仍是原值）。 */
function sanitize(o){
  const fixed = [];
  const put = (k, val) => {
    o[k] = val;
    /* 同步备份快照，避免修正后的值与快照不一致（否则回滚后又变回未修正的非法值）。
       类型判定：快照原值是「大数（__d 标记）」才写成大数，是普通数字就仍写数字。
       原实现一律写成 {m,e}，会把 number 字段变成大数对象，回滚后类型错乱。 */
    if(o._bak && typeof o._bak === 'object' && !BAK_SKIP.has(k)){
      /* 一律写修正后的「普通数字」原值 val：
         原实现在快照原值是大数（__d）时写成 {__d:1, m:String(val), e:'0'} ——
         e 被写死为 '0'，与 val 的真实量级无关；且回滚时 {m,e} 会把原本的 number 字段
         变成大数对象（level / kills 等），typeof x === 'number' 之类判断全部失效，类型错乱。
         经 sanitize 修正后的值必定是普通数字，直接写 val 即可，与 o[k] 类型完全一致。
         但若快照原值是「对象/数组型」({__s:1,j})，直接写数字会丢掉序列化包裹，
         快照类型错乱（将来若有代码按 __s 标记解析 _bak 就会踩坑），故按原格式同步。 */
      /* 一律写修正后的「普通数字」原值 val，不保留 __s / __d 包裹。
         原因：sanitize 只修正 SANE_MAX 里的数值字段，修正结果必定是普通数字（所有 SANE_MAX 都是数字）。
         若按快照原格式回写：
         · 原值是 __d（大数）→ 写成 {__d:1, m, e:'0'}，回滚时把该字段还原成大数对象，
           而 o[k] 此刻是普通数字 —— 类型错乱，S.kills++ 之类直接变 NaN（1.28 已修过一次，此处曾回退）；
         · 原值是 __s（对象/数组）→ 把数字塞进序列化包裹，回滚后得到一个「看起来是对象」的数字，同样错乱。
         快照的作用只是「提供上次合法值」，既然本次修正后的合法值已是数字，快照就该是同一个数字。 */
      o._bak[k] = val;
    }
  };
  for(const k in SANE_MAX){
    if(o[k] === undefined || o[k] === null) continue;
    let v = o[k];
    if(v && typeof v === 'object' && v.m !== undefined){
      /* 大数转 Number：e 过大时 Math.pow 直接返回 Infinity，不能简单判为非法。
         e > 300 表示 ≥1e300，必然超过任何 SANE_MAX，直接封顶；e < -300 视为 0。 */
      const e = Number(v.e);
      if(!isFinite(e) || e > 300){ put(k, SANE_MAX[k]); fixed.push(k); continue; }
      if(e < -300){ put(k, 0); fixed.push(k); continue; }
      v = Number(v.m) * Math.pow(10, e);
    }
    v = Number(v);
    if(!isFinite(v) || Number.isNaN(v)){ put(k, 0); fixed.push(k); continue; }
    if(v < 0){ put(k, 0); fixed.push(k); continue; }
    if(v > SANE_MAX[k]){ put(k, SANE_MAX[k]); fixed.push(k); }
  }
  return fixed;
}

function load(){
  try{
    const o = JSON.parse(localStorage.getItem('xiantu_save'), bigRev);
    /* ---- 轻量防篡改校验 ---- */
    if(o){
      /* 顺序很重要：必须先校验签名，再修正数值。
         若先 sanitize 再校验，修正后的数值会与原签名不符，导致「合法但超限」的存档被误判为篡改。 */
      let tampered = false, bad = [];
      if(o._sig === undefined || o._sig === null){
        bad = sanitize(o);                     /* 老存档：先修掉负数 / NaN / 天文数字，再静默补签 */
        o._sig = sigOf(o);
      } else if(o._sig !== sigOf(o)){
        /* 先按新格式算一份摘要留存：用于区分「真被篡改」与「签名格式升级」。
           （1.26 起 serVal 对数组改用独立的 [ ] 包裹 + 数值下标顺序，旧档签名必然对不上，
             但存档数值本身是好的，不该弹「存档异常」吓玩家。） */
        const sigLoaded = sigOf(o);
        tampered = true;                       /* 签名不符：数值被改过 */
        /* 用备份快照回滚被改的关键数值（备份本身也在签名保护内） */
        if(o._bak && typeof o._bak === 'object'){
          for(const k in o._bak){
            const v = o._bak[k];
            /* 与 save() 的写入格式对应：__d 标记为大数，其余按原类型还原（含对象深拷贝） */
            if(v && typeof v === 'object' && v.__d){
              /* e 字段可能被破坏成非数字字符串：BigInt() 抛错会跳过整个 load()，
                 玩家看到的是「存档没了」而非「检测到篡改」——必须逐字段兜底。 */
              try{
                const _m = Number(v.m), _e = BigInt(String(v.e));
                if(isFinite(_m)) o[k] = { m:_m, e:_e };
                else o[k] = 0;
              }catch(e){ o[k] = 0; }
            }
            /* 对象/数组型快照：必须用 bigRev 还原器。
               原写法 JSON.parse(v.j) 缺还原器 —— 存入时 bigRepl 把 BigInt 转成了 '\u0001数字' 字符串，
               parse 后仍是字符串，e 字段无法参与 BigInt 运算（dcmp / dadd 全部崩）。
               丹药改大数存储后 S.pills 走的就是这条 __s 分支，此修复是必需的。 */
            else if(v && typeof v === 'object' && v.__s){ try{ o[k] = JSON.parse(v.j, bigRev); }catch(e){ o[k] = v.j; } }
            else if(v && typeof v === 'object'){ try{ o[k] = JSON.parse(JSON.stringify(v)); }catch(e){ o[k] = v; } }   /* 旧档兼容：对象形态 */
            else o[k] = v;
          }
